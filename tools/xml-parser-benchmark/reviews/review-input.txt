    // C callbacks capture every exception: no C++ exception crosses libxml2 frames.
    struct LibxmlState
    {
      XMLHandler* handler;
      std::exception_ptr exception;
      static LibxmlState& state(void* ctx)
      {
        return *static_cast<LibxmlState*>(static_cast<xmlParserCtxt*>(ctx)->_private);
      }
      static std::u16string utf16(const xmlChar* text, int length = -1)
      {
        std::u16string result;
        const unsigned char* cursor = text;
        const unsigned char* end = text + (length < 0 ? xmlStrlen(text) : length);
        while (cursor < end)
        {
          unsigned int cp = *cursor++;
          if (cp >= 0xc0)
          {
            const int remaining = cp < 0xe0 ? 1 : cp < 0xf0 ? 2 : 3;
            cp &= (1u << (6 - remaining)) - 1;
            for (int i = 0; i < remaining && cursor < end; ++i) { cp = (cp << 6) | (*cursor++ & 0x3f); }
          }
          if (cp > 0xffff)
          {
            cp -= 0x10000;
            result.push_back(static_cast<char16_t>(0xd800 + (cp >> 10)));
            result.push_back(static_cast<char16_t>(0xdc00 + (cp & 0x3ff)));
          }
          else { result.push_back(static_cast<char16_t>(cp)); }
        }
        return result;
      }
      template <typename Callback> static void callback(void* ctx, Callback&& fn)
      {
        try { fn(state(ctx)); }
        catch (...)
        {
          state(ctx).exception = std::current_exception();
          xmlStopParser(static_cast<xmlParserCtxt*>(ctx));
        }
      }
      static void start(void* ctx, const xmlChar* name, const xmlChar** attrs)
      {
        callback(ctx, [&](LibxmlState& s)
        {
          std::vector<std::pair<std::u16string, std::u16string>> values;
          if (attrs)
          {
            for (Size i = 0; attrs[i]; i += 2) { values.emplace_back(utf16(attrs[i]), utf16(attrs[i + 1])); }
          }
          s.handler->onStartElement(utf16(name).c_str(), XMLAttributes(values));
        });
      }
      static void end(void* ctx, const xmlChar* name)
      {
        callback(ctx, [&](LibxmlState& s) { s.handler->onEndElement(utf16(name).c_str()); });
      }
      static void characters(void* ctx, const xmlChar* text, int length)
      {
        callback(ctx, [&](LibxmlState& s)
        {
          const auto value = utf16(text, length);
          s.handler->onCharacters(value.c_str(), value.size());
        });
      }
      static void error(void* ctx, const xmlError* error)
      {
        callback(ctx, [&](LibxmlState& s)
        {
          const std::string message = error->message ? error->message : "libxml2 parse error";
          if (error->level == XML_ERR_WARNING) { s.handler->warning(XMLHandler::LOAD, message, error->line, error->int2); }
          else { s.handler->fatalError(XMLHandler::LOAD, message, error->line, error->int2); }
        });
      }
    };
    void parse(xercesc::InputSource* const source, XMLHandler* handler)
    {
      LibxmlState state{handler, {}};
      xmlSAXHandler sax{};
      xmlSAXVersion(&sax, 1);
      sax.startElement = LibxmlState::start;
      sax.endElement = LibxmlState::end;
      sax.characters = LibxmlState::characters;
      sax.cdataBlock = LibxmlState::characters;
      sax.serror = LibxmlState::error;
      auto free_parser = [](xmlParserCtxt* ctx)
      {
        if (ctx->myDoc) { xmlFreeDoc(ctx->myDoc); }
        xmlFreeParserCtxt(ctx);
      };
      unique_ptr<xmlParserCtxt, decltype(free_parser)> parser(xmlCreatePushParserCtxt(&sax, nullptr, nullptr, 0, nullptr), free_parser);
      if (!parser) { throw std::bad_alloc(); }
      parser->_private = &state;
      parser->userData = parser.get();
      xmlCtxtUseOptions(parser.get(), XML_PARSE_NONET | XML_PARSE_NO_XXE | XML_PARSE_NOENT);
      if (source->getEncoding())
      {
        const auto encoding = StringManager::convert(reinterpret_cast<const char16_t*>(source->getEncoding()));
        if (xmlSwitchEncodingName(parser.get(), encoding.c_str()) != 0)
        {
          throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", "Unsupported XML encoding: " + encoding);
        }
      }
      unique_ptr<xercesc::BinInputStream> stream(source->makeStream());
      if (!stream) { throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", "Cannot open XML input stream"); }
      char buffer[65536];
      int status = 0;
      while (!state.exception)
      {
        const auto count = stream->readBytes(reinterpret_cast<XMLByte*>(buffer), sizeof(buffer));
        status = xmlParseChunk(parser.get(), buffer, static_cast<int>(count), count == 0);
        if (status || count == 0) { break; }
      }
      if (state.exception)
      {
        try { std::rethrow_exception(state.exception); }
        catch (const XMLHandler::EndParsingSoftly&) { return; }
      }
      if (status)
      {
        throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", "libxml2 rejected XML input");
      }
    }

    void XMLFile::parse_(const std::string & filename, XMLHandler * handler)
    {
      // ensure handler->reset() is called to save memory (in case the XMLFile
      // reader, e.g. FeatureXMLFile, is used again)
      XMLCleaner_ clean(handler);

      StringManager sm;
      //try to open file
      if (!File::exists(filename))
      {
        throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }

      // initialize parser
      try
      {
        Internal::xercesInitialize(); // serialised: Xerces requires it, and parseBuffer_() runs on several threads at once
      }
      catch (const xercesc::XMLException & toCatch)
      {
        throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            "",std::string("Error during initialization: ") + StringManager().convert(toCatch.getMessage()));
      }


      // peak ahead into the file: is it bzip2 or gzip compressed?
      std::string bz;
      {
        std::ifstream file(filename.c_str());
        char tmp_bz[3];
        file.read(tmp_bz, 2);
        tmp_bz[2] = '\0';
        bz =std::string(tmp_bz);
      }

      unique_ptr<xercesc::InputSource> source;

      char g1 = 0x1f;
      char g2 = 0;
      g2 |= 1 << 7;
      g2 |= 1 << 3;
      g2 |= 1 << 1;
      g2 |= 1 << 0;
      //g2 = static_cast<char>(0x8b); // can make troubles if it is casted to 0x7F which is the biggest number signed char can save
      if ((bz[0] == 'B' && bz[1] == 'Z') || (bz[0] == g1 && bz[1] == g2) || (bz[0] == 'P' && bz[1] == 'K'))
      {
        source.reset(new CompressedInputSource(sm.convert(filename).c_str(), bz));
      }
      else
      {
        source.reset(new xercesc::LocalFileInputSource(sm.convert(filename).c_str()));
      }
      // what if no encoding given http://xerces.apache.org/xerces-c/apiDocs-3/classInputSource.html
      if (!enforced_encoding_.empty())
      {
        const auto encoding = sm.convert(enforced_encoding_);
        source->setEncoding(reinterpret_cast<const XMLCh*>(encoding.c_str()));
      }
      
      parse(source.get(), handler);
    }

    void XMLFile::save_(const std::string & filename, XMLHandler * handler) const
    {
      // Detect compression from filename extension, in any letter case like on input (and in TOPPBase's check)
      const FileTypes::Type compression = FileNameUtils::compressionType(filename);
      const bool use_gzip = compression == FileTypes::GZ;
      const bool use_bzip2 = compression == FileTypes::BZ2;

      if (use_gzip)
      {
        // Minimal std::streambuf that writes gzip-compressed data via zlib
        class GzipOStreambuf : public std::streambuf
        {
        public:
          enum { BUF_SIZE = 65536 };

          explicit GzipOStreambuf(const char* fn) : gz_(gzopen(fn, "wb"))
          {
            if (gz_) setp(buf_, buf_ + BUF_SIZE);
          }

          ~GzipOStreambuf()
          {
            flush_buffer();
            if (gz_) gzclose(gz_);
          }

          bool is_open() const { return gz_ != nullptr; }

        protected:
          int_type overflow(int_type c) override
          {
            if (!gz_) return traits_type::eof();
            if (pptr() == epptr() && !flush_buffer()) return traits_type::eof();
            if (!traits_type::eq_int_type(c, traits_type::eof()))
            {
              *pptr() = traits_type::to_char_type(c);
              pbump(1);
            }
            return traits_type::not_eof(c);
          }

          int sync() override { return flush_buffer() ? 0 : -1; }

        private:
          bool flush_buffer()
          {
            int n = static_cast<int>(pptr() - pbase());
            if (n > 0)
            {
              if (gzwrite(gz_, pbase(), static_cast<unsigned>(n)) != n) return false;
              setp(buf_, buf_ + BUF_SIZE);
            }
            return true;
          }

          gzFile gz_;
          char buf_[BUF_SIZE];
        };

        GzipOStreambuf sbuf(filename.c_str());
        if (!sbuf.is_open())
        {
          throw Exception::UnableToCreateFile(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
        }
        std::ostream os(&sbuf);
        os.precision(writtenDigits(double()));
        handler->writeTo(os);
      }
      else if (use_bzip2)
      {
        // Minimal std::streambuf that writes bzip2-compressed data via bzlib
        class Bzip2OStreambuf : public std::streambuf
        {
        public:
          enum { BUF_SIZE = 65536 };

          explicit Bzip2OStreambuf(const char* fn) : file_(nullptr), bz_(nullptr)
          {
            file_ = fopen(fn, "wb");
            if (!file_) return;
            int err = BZ_OK;
            bz_ = BZ2_bzWriteOpen(&err, file_, 9, 0, 0);
            if (err != BZ_OK)
            {
              BZ2_bzWriteClose(&err, bz_, 1, nullptr, nullptr);
              bz_ = nullptr;
              fclose(file_);
              file_ = nullptr;
              return;
            }
            setp(buf_, buf_ + BUF_SIZE);
          }

          ~Bzip2OStreambuf()
          {
            flush_buffer();
            if (bz_)
            {
              int err = BZ_OK;
              BZ2_bzWriteClose(&err, bz_, 0, nullptr, nullptr);
            }
            if (file_) fclose(file_);
          }

          bool is_open() const { return bz_ != nullptr; }

        protected:
          int_type overflow(int_type c) override
          {
            if (!bz_) return traits_type::eof();
            if (pptr() == epptr() && !flush_buffer()) return traits_type::eof();
            if (!traits_type::eq_int_type(c, traits_type::eof()))
            {
              *pptr() = traits_type::to_char_type(c);
              pbump(1);
            }
            return traits_type::not_eof(c);
          }

          int sync() override { return flush_buffer() ? 0 : -1; }

        private:
          bool flush_buffer()
          {
            if (!bz_) return false;
            int n = static_cast<int>(pptr() - pbase());
            if (n > 0)
            {
              int err = BZ_OK;
              BZ2_bzWrite(&err, bz_, pbase(), n);
              if (err != BZ_OK && err != BZ_RUN_OK) return false;
              setp(buf_, buf_ + BUF_SIZE);
            }
            return true;
          }

          FILE* file_;
          BZFILE* bz_;
          char buf_[BUF_SIZE];
        };

        Bzip2OStreambuf sbuf(filename.c_str());
        if (!sbuf.is_open())
        {
          throw Exception::UnableToCreateFile(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
        }
        std::ostream os(&sbuf);
        os.precision(writtenDigits(double()));
        handler->writeTo(os);
      }
      else
      {
        // Uncompressed: open in binary mode to avoid any line ending conversions
        std::ofstream os(filename.c_str(), std::ios::out | std::ios::binary);
        os.precision(writtenDigits(double()));
        if (!os)
        {
          throw Exception::UnableToCreateFile(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
        }
        handler->writeTo(os);
        os.close();
      }
    }

  {
    // Error if intensity or m/z (RT) is encoded as int32|64 - they should be float32|64!

    Size int_size = int_precision_64 ? data[int_index].floats_64.size() : data[int_index].floats_32.size();

    if (mz_size != int_size)

        {
          fillDataArrayString(input_data[i], spectrum);
        }
      }
    }
  }

  inline void fillDataArrays(const std::vector<Internal::MzMLHandlerHelper::BinaryData>& input_data, MSChromatogram& spectrum)
  {
    for (Size i = 0; i < input_data.size(); i++) //loop over all binary data arrays
    {
      if (input_data[i].meta.getName() != "time array" && input_data[i].meta.getName() != "intensity array") // is meta data array?
      {
        if (input_data[i].data_type == Internal::MzMLHandlerHelper::BinaryData::DT_FLOAT)
        {
          fillDataArrayFloat(input_data[i], spectrum);
        }
        else if (input_data[i].data_type == Internal::MzMLHandlerHelper::BinaryData::DT_INT)
        {
          fillDataArrayInt(input_data[i], spectrum);
        }
        else if (input_data[i].data_type == Internal::MzMLHandlerHelper::BinaryData::DT_STRING)
        {
          fillDataArrayString(input_data[i], spectrum);
        }
      }
    }
  }

  template<class T>
  inline void fillDataArray(const std::vector<Internal::MzMLHandlerHelper::BinaryData>& data,
                            T& peak_container,
                            const bool x_precision_64,
                            const bool int_precision_64,
                            const SignedSize x_index,
                            const SignedSize int_index,
                            const Size default_array_length_)
  {
    typename T::PeakType tmp;
    if (x_precision_64 && !int_precision_64)
    {
      std::vector< double >::const_iterator mz_it = data[x_index].floats_64.begin();
      std::vector< float >::const_iterator int_it = data[int_index].floats_32.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
    else if (x_precision_64 && int_precision_64)
    {
      std::vector< double >::const_iterator mz_it = data[x_index].floats_64.begin();
      std::vector< double >::const_iterator int_it = data[int_index].floats_64.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
    else if (!x_precision_64 && int_precision_64)
    {
      std::vector< float >::const_iterator mz_it = data[x_index].floats_32.begin();
      std::vector< double >::const_iterator int_it = data[int_index].floats_64.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
    else if (!x_precision_64 && !int_precision_64)
    {
      std::vector< float >::const_iterator mz_it = data[x_index].floats_32.begin();
      std::vector< float >::const_iterator int_it = data[int_index].floats_32.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
  }

  void MzMLSpectrumDecoder::handleBinaryDataArray_(void* opaque, std::vector<BinaryData>& data)
  {
    auto* array = static_cast<xmlNode*>(opaque);
    data.emplace_back();
    bool has_binary = false;
    for (xmlNode* node = array->children; node; node = node->next)
    {
      if (node->type != XML_ELEMENT_NODE) { continue; }
      if (xmlStrEqual(node->name, BAD_CAST "binary"))
      {
        has_binary = true;
        if (!node->children) { continue; }
        if (node->children->next || node->children->type != XML_TEXT_NODE)
        {
          throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            "", "Invalid XML: 'binary' element can only have a single, text node child element.");
        }
        data.back().base64.append(reinterpret_cast<const char*>(node->children->content));
      }
      else if (xmlStrEqual(node->name, BAD_CAST "cvParam"))
      {
        Internal::MzMLHandlerHelper::handleBinaryDataArrayCVParam(data, xmlProperty(node, "accession"),
          xmlProperty(node, "value"), xmlProperty(node, "name"), xmlProperty(node, "unitAccession"));
      }
      else if (xmlStrEqual(node->name, BAD_CAST "userParam")) { std::cout << " unhandled userParam\n"; }
      else if (xmlStrEqual(node->name, BAD_CAST "referenceableParamGroupRef")) { std::cout << " unhandled referenceableParamGroupRef\n"; }
    }
    if (!has_binary)
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
        "", "Invalid XML: 'binary' element needs to be present at least once inside 'binaryDataArray' element.");
    }
  }

  std::string MzMLSpectrumDecoder::domParseString_(const std::string& in, std::vector<BinaryData>& data)
  {
    if (in.size() > static_cast<Size>(std::numeric_limits<int>::max()))
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", "XML fragment exceeds libxml2 memory API size limit");
    }
    std::unique_ptr<xmlParserCtxt, decltype(&xmlFreeParserCtxt)> parser(xmlNewParserCtxt(), xmlFreeParserCtxt);
    if (!parser) { throw std::bad_alloc(); }
    // Large spectra can have a base64 text node exceeding the default 10 MB cap.
    // HUGE also raises depth/name caps; external entities stay disabled and the
    // independent entity-amplification limit is explicitly retained at 5.
    xmlCtxtSetMaxAmplification(parser.get(), 5);
    // Indexed offset spans include the completed root followed by parent closing
    // tags (and sometimes the next list opening). Stop at the real XML root end,
    // using the parser rather than a lexical search through comments or CDATA.
    xmlDoc* completed_document = nullptr;
    parser->_private = &completed_document;
    parser->sax->endElementNs = [](void* opaque, const xmlChar* name, const xmlChar* prefix, const xmlChar* uri)
    {
      auto* ctxt = static_cast<xmlParserCtxt*>(opaque);
      xmlSAX2EndElementNs(ctxt, name, prefix, uri);
      if (!ctxt->node)
      {
        *static_cast<xmlDoc**>(ctxt->_private) = ctxt->myDoc;
        ctxt->myDoc = nullptr; // transfer ownership before xmlCtxtReadMemory handles USER_STOP
        xmlStopParser(ctxt);
      }
    };
    xmlDoc* returned_document = xmlCtxtReadMemory(parser.get(), in.data(), static_cast<int>(in.size()), "spectrum.xml", nullptr,
      XML_PARSE_NONET | XML_PARSE_NO_XXE | XML_PARSE_NOENT | XML_PARSE_HUGE | XML_PARSE_NOERROR | XML_PARSE_NOWARNING);
    std::unique_ptr<xmlDoc, decltype(&xmlFreeDoc)> document(completed_document ? completed_document : returned_document, xmlFreeDoc);
    xmlNode* root = document ? xmlDocGetRootElement(document.get()) : nullptr;
    if (!root) { throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, in, "No root element or malformed XML"); }
    OPENMS_PRECONDITION(xmlStrEqual(root->name, BAD_CAST "spectrum") || xmlStrEqual(root->name, BAD_CAST "chromatogram"),
      "The input needs to contain a <spectrum> or <chromatogram> tag as root element.")
    if (!xmlHasProp(root, BAD_CAST "defaultArrayLength"))
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, in, "Root element does not contain defaultArrayLength XML tag.");
    }
    const int length = StringUtils::toInt32(xmlProperty(root, "defaultArrayLength"));
    // Depth-first traversal replaces Xerces getElementsByTagName without a second list.
    for (xmlNode* node = root->children; node;)
    {
      if (node->type == XML_ELEMENT_NODE && xmlStrEqual(node->name, BAD_CAST "binaryDataArray"))
      {
        handleBinaryDataArray_(node, data);
        data.back().size = length;
      }
      if (node->children) { node = node->children; continue; }
      while (node != root && !node->next) { node = node->parent; }
      if (node == root) { break; }
      node = node->next;
    }
    return xmlProperty(root, "id");
  }

    std::streampos stringToStreampos(const std::string& s)
    {
      // Try to cast the string to a type that can hold the integer value
      // stored in the std::streampos type (which can vary from 32 to 128 bits
      // depending on the system).
      //
      // long long has a minimal size of 64 bits and can address a range up to
      // 16 Exbibit (or 2 Exbibyte), we can hopefully expect our files to be
      // smaller than an Exabyte and should be safe.
      std::streampos res;
      try
      {

        res = boost::lexical_cast< unsigned long long >(s);
        // TESTING CAST: res = (int) res; // use this only when emulating 32 bit systems

      }
      catch (boost::bad_lexical_cast&)
      {
        std::cerr << "Trying to convert corrupted / unreadable value to std::streampos : " << s << std::endl;
        std::cerr << "This can also happen if the value exceeds 63 bits, please check your input." << std::endl;
        throw Exception::ConversionError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            std::string("Could not convert string '") + s + "' to a 64 bit integer.");
      }

      // Check if the value can fit into std::streampos
      if ( std::abs( boost::lexical_cast< long double >(s) - res) > 0.1)
      {
        std::cerr << "Your system may not support addressing a file of this size,"
          << " only addresses that fit into a " << sizeof(std::streamsize)*8 <<
          " bit integer are supported on your system." << std::endl;
        throw Exception::ConversionError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            std::string("Could not convert string '") + s + "' to an integer on your system.");
      }

      return res;
    }

  int IndexedMzMLDecoder::parseOffsets(const std::string& filename, std::streampos indexoffset, OffsetVector& spectra_offsets, OffsetVector& chromatograms_offsets)
  {
    //-------------------------------------------------------------
    // Open file, jump to end and read last indexoffset bytes into buffer.
    //-------------------------------------------------------------
    std::ifstream f(filename.c_str());
    if (!f.is_open())
    {
      if (!File::exists(filename))
      {
        throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else if (!File::readable(filename))
      {
        throw Exception::FileNotReadable(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else
      {
        throw Exception::IOException(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
    }

    // get length of file:
    f.seekg(0, f.end);
    std::streampos length = f.tellg();

    if (indexoffset < 0 || indexoffset > length)
    {
      std::cerr << "IndexedMzMLDecoder::parseOffsets Error: Offset was " <<
        indexoffset << " (not between 0 and " << length << ")." << std::endl;
      return -1;
    }

    //-------------------------------------------------------------
    // Read full end of file to parse offsets for spectra and chroms
    //-------------------------------------------------------------
    // read data as a block into a buffer

    // allocate enough memory in buffer (+1 for string termination)
    std::streampos readl = length - indexoffset;
    char* buffer = new(std::nothrow) char[readl + std::streampos(1)];

    // catch case where not enough memory is available
    if (buffer == nullptr)
    {
      // Warning: Index takes up more than 10 % of the whole file, please check your input file.\n";
      std::cerr << "IndexedMzMLDecoder::parseOffsets Could not allocate enough memory to read in index of indexedMzML" << std::endl;
      std::cerr << "IndexedMzMLDecoder::parseOffsets calculated index offset " << indexoffset << " and file length " << length <<
        ", consequently tried to read into memory " << readl << " bytes." << std::endl;
      return -1;
    }

    // read into memory
    f.seekg(-readl, f.end);
    f.read(buffer, readl);
    buffer[readl] = '\0';

    //-------------------------------------------------------------
    // Add a sane start element and then give it to a DOM parser
    //-------------------------------------------------------------
    // http://stackoverflow.com/questions/4691039/making-xerces-parse-a-string-insted-of-a-file
    std::string tmp_fixed_xml = "<indexedmzML>" +  std::string(buffer) + "\n";
    int res = domParseIndexedEnd_(tmp_fixed_xml, spectra_offsets, chromatograms_offsets);

    delete[] buffer;

    return res;
  }

  std::streampos IndexedMzMLDecoder::findIndexListOffset(const std::string& filename, int buffersize)
  {
    // return value
    std::streampos indexoffset = -1;

    //-------------------------------------------------------------
    // Open file, jump to end and read last n bytes into buffer.
    //-------------------------------------------------------------
    std::ifstream f(filename.c_str());

    if (!f.is_open())
    {
      if (!File::exists(filename))
      {
        throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else if (!File::readable(filename))
      {
        throw Exception::FileNotReadable(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else
      {
        throw Exception::IOException(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
    }

    // Read the last few bytes and hope our offset is there to be found
    std::unique_ptr<char[]> buffer(new char[buffersize + 1]);
    f.seekg(-buffersize, f.end);
    f.read(buffer.get(), buffersize);
    buffer.get()[buffersize] = '\0';

#ifdef DEBUG_READER
    std::cout << " reading file " << filename  << " with size " << buffersize << '\n';
    std::cout << buffer << '\n';
#endif

    //-------------------------------------------------------------
    // Since we could be anywhere in the XML structure, use regex to find
    // indexListOffset and read its content.
    //-------------------------------------------------------------
    boost::regex listoffset_rx(R"(<[^>/]*indexListOffset\s*>\s*(\d*))");
    boost::cmatch matches;
    boost::regex_search(buffer.get(), matches, listoffset_rx);
    std::string thismatch(matches[1].first, matches[1].second);
    if (!thismatch.empty())
    {
      try
      {
        indexoffset = OpenMS::IndexedMzMLUtils::stringToStreampos(thismatch);
      }
      catch (Exception::ConversionError& /*e*/)
      {
        std::cerr << "Corrupted / unreadable value in <indexListOffset> : " << thismatch << std::endl;
        // free resources and re-throw
        throw;  // re-throw conversion error
      }
    }
    else
    {
      std::cerr << "IndexedMzMLDecoder::findIndexListOffset Error: Could not find element indexListOffset in the last "
                << buffersize << " bytes. Maybe this is not a indexedMzML."
                << buffer.get() << std::endl;
    }

    return indexoffset;
  }

  int IndexedMzMLDecoder::domParseIndexedEnd_(const std::string& in, OffsetVector& spectra_offsets, OffsetVector& chromatograms_offsets)
  {

    if (in.size() > static_cast<Size>(std::numeric_limits<int>::max())) { return -1; }
    std::unique_ptr<xmlDoc, decltype(&xmlFreeDoc)> document(
      xmlReadMemory(in.data(), static_cast<int>(in.size()), "index.xml", nullptr,
                    XML_PARSE_NONET | XML_PARSE_NO_XXE | XML_PARSE_NOENT | XML_PARSE_NOERROR | XML_PARSE_NOWARNING), xmlFreeDoc);
    xmlNode* root = document ? xmlDocGetRootElement(document.get()) : nullptr;
    if (!root) { return -1; }
    xmlNode* list = nullptr;
    for (xmlNode* node = root->children; node; node = node->next)
    {
      if (node->type == XML_ELEMENT_NODE && xmlStrEqual(node->name, BAD_CAST "indexList"))
      {
        if (list) { return -1; }
        list = node;
      }
    }
    if (!list) { return -1; }
    auto property = [](xmlNode* node, const char* key)
    {
      xmlChar* value = xmlGetProp(node, BAD_CAST key);
      std::string result = value ? reinterpret_cast<const char*>(value) : "";
      xmlFree(value);
      return result;
    };
    for (xmlNode* index = list->children; index; index = index->next)
    {
      if (index->type != XML_ELEMENT_NODE) { continue; }
      OffsetVector result;
      for (xmlNode* offset = index->children; offset; offset = offset->next)
      {
        if (offset->type != XML_ELEMENT_NODE) { continue; }
        xmlChar* content = xmlNodeGetContent(offset);
        std::string text = content ? reinterpret_cast<const char*>(content) : "";
        xmlFree(content);
        result.emplace_back(property(offset, "idRef"), IndexedMzMLUtils::stringToStreampos(text));
      }
      const auto name = property(index, "name");
      if (name == "spectrum") { spectra_offsets = std::move(result); }
      else if (name == "chromatogram") { chromatograms_offsets = std::move(result); }
      else { return -1; }
    }
    return 0;
  }

      static_assert(sizeof(XML_Char) == 1, "This adapter requires UTF-8 Expat callbacks.");

      // Expat validates UTF-8 before callbacks; OpenMS handlers still consume UTF-16.
      std::u16string utf16(std::string_view text)
      {
        std::u16string result;
        result.reserve(text.size());
        for (Size i = 0; i < text.size();)
        {
          unsigned char lead = static_cast<unsigned char>(text[i++]);
          unsigned code = lead;
          unsigned continuation = 0;
          if (lead >= 0xf0) { code &= 0x07; continuation = 3; }
          else if (lead >= 0xe0) { code &= 0x0f; continuation = 2; }
          else if (lead >= 0xc0) { code &= 0x1f; continuation = 1; }
          for (unsigned j = 0; j < continuation; ++j)
          {
            code = (code << 6) | (static_cast<unsigned char>(text[i++]) & 0x3f);
          }
          if (code > 0xffff)
          {
            code -= 0x10000;
            result.push_back(static_cast<char16_t>(0xd800 + (code >> 10)));
            result.push_back(static_cast<char16_t>(0xdc00 + (code & 0x3ff)));
          }
          else result.push_back(static_cast<char16_t>(code));
        }
        return result;
      }

      struct ExpatCallbacks
      {
        XML_Parser parser;
        XMLHandler& handler;
        std::exception_ptr exception;

        template <typename Callback>
        void invoke(Callback&& callback) noexcept
        {
          // Never unwind a C++ exception through Expat's C stack.
          if (exception) return;
          try { callback(); }
          catch (...)
          {
            exception = std::current_exception();
            XML_StopParser(parser, XML_FALSE);
          }
        }

        static void XMLCALL start(void* data, const XML_Char* name, const XML_Char** attributes) noexcept
        {
          auto& self = *static_cast<ExpatCallbacks*>(data);
          self.invoke([&]
          {
            const auto tag = utf16(name);
            std::vector<std::u16string> strings;
            for (Size i = 0; attributes[i]; ++i) strings.push_back(utf16(attributes[i]));
            std::vector<const char16_t*> pairs;
            pairs.reserve(strings.size() + 1);
            for (const auto& value : strings) pairs.push_back(value.c_str());
            pairs.push_back(nullptr);
            self.handler.onStartElement(tag.c_str(), XMLAttributes(pairs.data()));
          });
        }

        static void XMLCALL end(void* data, const XML_Char* name) noexcept
        {
          auto& self = *static_cast<ExpatCallbacks*>(data);
          self.invoke([&]
          {
            const auto tag = utf16(name);
            self.handler.onEndElement(tag.c_str());
          });
        }

        static void XMLCALL characters(void* data, const XML_Char* text, int length) noexcept
        {
          auto& self = *static_cast<ExpatCallbacks*>(data);
          self.invoke([&]
          {
            const auto value = utf16(std::string_view(text, static_cast<Size>(length)));
            self.handler.onCharacters(value.c_str(), value.size());
          });
        }
      };
    }

    void parse(xercesc::InputSource* source, XMLHandler* handler, const std::string& encoding)
    {
      // Reuse the existing bounded-memory plain/compressed streams. Xerces remains
      // a dependency for these stream adapters, DOM readers and XSD validation.
      unique_ptr<XML_ParserStruct, decltype(&XML_ParserFree)> parser(
        XML_ParserCreate(encoding.empty() ? nullptr : encoding.c_str()), XML_ParserFree);
      if (!parser) throw std::bad_alloc();
      ExpatCallbacks callbacks{parser.get(), *handler, {}};
      XML_SetUserData(parser.get(), &callbacks);
      XML_SetElementHandler(parser.get(), ExpatCallbacks::start, ExpatCallbacks::end);
      XML_SetCharacterDataHandler(parser.get(), ExpatCallbacks::characters);
      // Ordinary OpenMS SAX reads use qualified names without namespace expansion.
      // External entities are rejected; parameter entity parsing is disabled.
      XML_SetParamEntityParsing(parser.get(), XML_PARAM_ENTITY_PARSING_NEVER);
      XML_SetExternalEntityRefHandler(parser.get(), [](XML_Parser, const XML_Char*, const XML_Char*, const XML_Char*, const XML_Char*) -> int { return XML_STATUS_ERROR; });
      try
      {
        unique_ptr<xercesc::BinInputStream> stream(source->makeStream());
        if (!stream) throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", "Cannot open XML input stream");
        std::array<XMLByte, 65536> buffer;
        for (;;)
        {
          const auto length = stream->readBytes(buffer.data(), buffer.size());
          const auto status = XML_Parse(parser.get(), reinterpret_cast<const char*>(buffer.data()), static_cast<int>(length), length == 0);
          if (callbacks.exception) std::rethrow_exception(callbacks.exception);
          if (status == XML_STATUS_ERROR)
          {
            handler->fatalError(XMLHandler::LOAD, XML_ErrorString(XML_GetErrorCode(parser.get())),
                                static_cast<UInt>(XML_GetCurrentLineNumber(parser.get())),
                                static_cast<UInt>(XML_GetCurrentColumnNumber(parser.get()) + 1));
          }
          if (length == 0) break;
        }
      }
      catch (const xercesc::XMLException& error)
      {
        throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", StringManager::convert(error.getMessage()));
      }
      catch (const XMLHandler::EndParsingSoftly&)
      {
        // Expected early termination (metadata/size-only reads).
      }
    }

    void XMLFile::parse_(const std::string & filename, XMLHandler * handler)
    {
      // ensure handler->reset() is called to save memory (in case the XMLFile
      // reader, e.g. FeatureXMLFile, is used again)
      XMLCleaner_ clean(handler);

      StringManager sm;
      //try to open file
      if (!File::exists(filename))
      {
        throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }

      // initialize parser
      try
      {
        Internal::xercesInitialize(); // serialised: Xerces requires it, and parseBuffer_() runs on several threads at once
      }
      catch (const xercesc::XMLException & toCatch)
      {
        throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            "",std::string("Error during initialization: ") + StringManager().convert(toCatch.getMessage()));
      }


      // peak ahead into the file: is it bzip2 or gzip compressed?
      std::string bz;
      {
        std::ifstream file(filename.c_str());
        char tmp_bz[3] = {};
        file.read(tmp_bz, 2);
        tmp_bz[2] = '\0';
        bz = std::string(tmp_bz, 2);
      }

      unique_ptr<xercesc::InputSource> source;

      char g1 = 0x1f;
      char g2 = 0;
      g2 |= 1 << 7;
      g2 |= 1 << 3;
      g2 |= 1 << 1;
      g2 |= 1 << 0;
      //g2 = static_cast<char>(0x8b); // can make troubles if it is casted to 0x7F which is the biggest number signed char can save
      if ((bz[0] == 'B' && bz[1] == 'Z') || (bz[0] == g1 && bz[1] == g2) || (bz[0] == 'P' && bz[1] == 'K'))
      {
        source.reset(new CompressedInputSource(sm.convert(filename).c_str(), bz));
      }
      else
      {
        source.reset(new xercesc::LocalFileInputSource(sm.convert(filename).c_str()));
      }
      parse(source.get(), handler, enforced_encoding_);
    }

    void XMLFile::save_(const std::string & filename, XMLHandler * handler) const
    {
      // Detect compression from filename extension, in any letter case like on input (and in TOPPBase's check)
      const FileTypes::Type compression = FileNameUtils::compressionType(filename);
      const bool use_gzip = compression == FileTypes::GZ;
      const bool use_bzip2 = compression == FileTypes::BZ2;

      if (use_gzip)
      {
        // Minimal std::streambuf that writes gzip-compressed data via zlib
        class GzipOStreambuf : public std::streambuf
        {
        public:
          enum { BUF_SIZE = 65536 };

          explicit GzipOStreambuf(const char* fn) : gz_(gzopen(fn, "wb"))
          {
            if (gz_) setp(buf_, buf_ + BUF_SIZE);
          }

          ~GzipOStreambuf()
          {
            flush_buffer();
            if (gz_) gzclose(gz_);
          }

          bool is_open() const { return gz_ != nullptr; }

        protected:
          int_type overflow(int_type c) override
          {
            if (!gz_) return traits_type::eof();
            if (pptr() == epptr() && !flush_buffer()) return traits_type::eof();
            if (!traits_type::eq_int_type(c, traits_type::eof()))
            {
              *pptr() = traits_type::to_char_type(c);
              pbump(1);
            }
            return traits_type::not_eof(c);
          }

          int sync() override { return flush_buffer() ? 0 : -1; }

        private:
          bool flush_buffer()
          {
            int n = static_cast<int>(pptr() - pbase());
            if (n > 0)
            {
              if (gzwrite(gz_, pbase(), static_cast<unsigned>(n)) != n) return false;
              setp(buf_, buf_ + BUF_SIZE);
            }
            return true;
          }

          gzFile gz_;
          char buf_[BUF_SIZE];
        };

        GzipOStreambuf sbuf(filename.c_str());
        if (!sbuf.is_open())
        {
          throw Exception::UnableToCreateFile(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
        }
        std::ostream os(&sbuf);
        os.precision(writtenDigits(double()));
        handler->writeTo(os);
      }
      else if (use_bzip2)
      {
        // Minimal std::streambuf that writes bzip2-compressed data via bzlib
        class Bzip2OStreambuf : public std::streambuf
        {
        public:
          enum { BUF_SIZE = 65536 };

          explicit Bzip2OStreambuf(const char* fn) : file_(nullptr), bz_(nullptr)
          {
            file_ = fopen(fn, "wb");
            if (!file_) return;
            int err = BZ_OK;
            bz_ = BZ2_bzWriteOpen(&err, file_, 9, 0, 0);
            if (err != BZ_OK)
            {
              BZ2_bzWriteClose(&err, bz_, 1, nullptr, nullptr);
              bz_ = nullptr;
              fclose(file_);
              file_ = nullptr;
              return;
            }
            setp(buf_, buf_ + BUF_SIZE);
          }

          ~Bzip2OStreambuf()
          {
            flush_buffer();
            if (bz_)
            {
              int err = BZ_OK;
              BZ2_bzWriteClose(&err, bz_, 0, nullptr, nullptr);
            }
            if (file_) fclose(file_);
          }

          bool is_open() const { return bz_ != nullptr; }

        protected:
          int_type overflow(int_type c) override
          {
            if (!bz_) return traits_type::eof();
            if (pptr() == epptr() && !flush_buffer()) return traits_type::eof();
            if (!traits_type::eq_int_type(c, traits_type::eof()))
            {
              *pptr() = traits_type::to_char_type(c);
              pbump(1);
            }
            return traits_type::not_eof(c);
          }

          int sync() override { return flush_buffer() ? 0 : -1; }

        private:
          bool flush_buffer()
          {
            if (!bz_) return false;
            int n = static_cast<int>(pptr() - pbase());
            if (n > 0)
            {
              int err = BZ_OK;
              BZ2_bzWrite(&err, bz_, pbase(), n);
              if (err != BZ_OK && err != BZ_RUN_OK) return false;
              setp(buf_, buf_ + BUF_SIZE);
            }
            return true;
          }

          FILE* file_;
          BZFILE* bz_;
          char buf_[BUF_SIZE];
        };

        Bzip2OStreambuf sbuf(filename.c_str());
        if (!sbuf.is_open())
        {
          throw Exception::UnableToCreateFile(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
        }
        std::ostream os(&sbuf);
        os.precision(writtenDigits(double()));
        handler->writeTo(os);
      }
      else
      {
        // Uncompressed: open in binary mode to avoid any line ending conversions
        std::ofstream os(filename.c_str(), std::ios::out | std::ios::binary);
        os.precision(writtenDigits(double()));
        if (!os)
        {
          throw Exception::UnableToCreateFile(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
        }
        handler->writeTo(os);
        os.close();
      }
    }

    std::string encodeTab(const std::string& to_encode)

  {

  inline void fillDataArrayString(const Internal::MzMLHandlerHelper::BinaryData& data, T& spectrum)
  {
    //create new array
    spectrum.getStringDataArrays().resize(spectrum.getStringDataArrays().size() + 1);
    //reserve space in the array
    spectrum.getStringDataArrays().back().reserve(data.decoded_char.size());
    //copy meta info into MetaInfoDescription
    spectrum.getStringDataArrays().back().MetaInfoDescription::operator=(data.meta);

    if (data.precision == Internal::MzMLHandlerHelper::BinaryData::PRE_64)
    {
      for (Size n = 0; n < data.decoded_char.size(); n++)
      {
        std::string value = data.decoded_char[n];
        spectrum.getStringDataArrays().back().push_back(value);
      }
    }
  }

  inline void fillDataArrays(const std::vector<Internal::MzMLHandlerHelper::BinaryData>& input_data, MSSpectrum& spectrum)
  {
    for (Size i = 0; i < input_data.size(); i++) //loop over all binary data arrays
    {
      if (input_data[i].meta.getName() != "m/z array" && input_data[i].meta.getName() != "intensity array") // is meta data array?
      {
        if (input_data[i].data_type == Internal::MzMLHandlerHelper::BinaryData::DT_FLOAT)
        {
          fillDataArrayFloat(input_data[i], spectrum);
        }
        else if (input_data[i].data_type == Internal::MzMLHandlerHelper::BinaryData::DT_INT)
        {
          fillDataArrayInt(input_data[i], spectrum);
        }
        else if (input_data[i].data_type == Internal::MzMLHandlerHelper::BinaryData::DT_STRING)
        {
          fillDataArrayString(input_data[i], spectrum);
        }
      }
    }
  }

  inline void fillDataArrays(const std::vector<Internal::MzMLHandlerHelper::BinaryData>& input_data, MSChromatogram& spectrum)
  {
    for (Size i = 0; i < input_data.size(); i++) //loop over all binary data arrays

      }
    }
  }

  template<class T>
  inline void fillDataArray(const std::vector<Internal::MzMLHandlerHelper::BinaryData>& data,
                            T& peak_container,
                            const bool x_precision_64,
                            const bool int_precision_64,
                            const SignedSize x_index,
                            const SignedSize int_index,
                            const Size default_array_length_)
  {
    typename T::PeakType tmp;
    if (x_precision_64 && !int_precision_64)
    {
      std::vector< double >::const_iterator mz_it = data[x_index].floats_64.begin();
      std::vector< float >::const_iterator int_it = data[int_index].floats_32.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
    else if (x_precision_64 && int_precision_64)
    {
      std::vector< double >::const_iterator mz_it = data[x_index].floats_64.begin();
      std::vector< double >::const_iterator int_it = data[int_index].floats_64.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
    else if (!x_precision_64 && int_precision_64)
    {
      std::vector< float >::const_iterator mz_it = data[x_index].floats_32.begin();
      std::vector< double >::const_iterator int_it = data[int_index].floats_64.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
    else if (!x_precision_64 && !int_precision_64)
    {
      std::vector< float >::const_iterator mz_it = data[x_index].floats_32.begin();
      std::vector< float >::const_iterator int_it = data[int_index].floats_32.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
  }

  void MzMLSpectrumDecoder::decodeBinaryDataMSSpectrum_(std::vector<BinaryData>& data, OpenMS::MSSpectrum& spectrum) const
  {
    Internal::MzMLHandlerHelper::decodeBase64Arrays(data, skip_xml_checks_);

    //look up the precision and the index of the intensity and m/z array
    bool x_precision_64 = true;
    bool int_precision_64 = true;
    SignedSize x_index = -1;
    SignedSize int_index = -1;
    Internal::MzMLHandlerHelper::computeDataProperties_(data, x_precision_64, x_index, "m/z array");
    Internal::MzMLHandlerHelper::computeDataProperties_(data, int_precision_64, int_index, "intensity array");

    //Abort if no m/z or intensity array is present
    if (int_index == -1 || x_index == -1)
    {
      std::cerr << "Error, intensity or m/z array is missing, skipping this spectrum" << std::endl;
      return;
    }

    checkData_(data, x_index, int_index, x_precision_64, int_precision_64);

    // At this point, we could check whether the defaultArrayLength from the
    // spectrum/chromatogram tag is equivalent to size of the decoded data
    Size default_array_length_ = x_precision_64 ? data[x_index].floats_64.size() : data[x_index].floats_32.size();

    spectrum.reserve(default_array_length_);
    fillDataArray(data, spectrum,
                  x_precision_64, int_precision_64,
                  x_index, int_index, default_array_length_);

    if (data.size() > 2) 
    {
      fillDataArrays(data, spectrum);
    }
  }

  void MzMLSpectrumDecoder::handleBinaryDataArray_(void* indexListNode_opaque, std::vector<BinaryData>& data)
  {
    xercesc::DOMNode* indexListNode = static_cast<xercesc::DOMNode*>(indexListNode_opaque);
    // access result through data.back()
    data.emplace_back();

    // using CONST_XMLCH since writing a u16 char array each time is ugly. disadvantage = no constexpr.
    // Uses only reinterpret_cast, so usually no runtime cost though.
    static const XMLCh* TAG_CV = CONST_XMLCH("cvParam");
    static const XMLCh* TAG_binary = CONST_XMLCH("binary");
    static const XMLCh* TAG_userParam = CONST_XMLCH("userParam");
    static const XMLCh* TAG_referenceableParamGroupRef = CONST_XMLCH("referenceableParamGroupRef");
    static const XMLCh* TAG_accession = CONST_XMLCH("accession");
    static const XMLCh* TAG_unit_accession = CONST_XMLCH("unitAccession");
    static const XMLCh* TAG_value = CONST_XMLCH("value");
    static const XMLCh* TAG_name = CONST_XMLCH("name");

    OpenMS::Internal::StringManager sm;

    // Iterate through binaryDataArray elements
    // only allowed subelements:
    //  - referenceableParamGroupRef (0+)
    //  - cvParam (0+)
    //  - userParam (0+)
    //  - binary (1)
    xercesc::DOMNodeList* index_elems = indexListNode->getChildNodes();
    const  XMLSize_t nodeCount_ = index_elems->getLength();
    bool has_binary_tag = false;
    for (XMLSize_t j = 0; j < nodeCount_; ++j)
    {
      xercesc::DOMNode* currentNode = index_elems->item(j);
      if (currentNode->getNodeType() &&   // true is not NULL
          currentNode->getNodeType() == xercesc::DOMNode::ELEMENT_NODE)   // is element
      {
        xercesc::DOMElement* currentElement = dynamic_cast<xercesc::DOMElement*>(currentNode);
        if (xercesc::XMLString::equals(currentElement->getTagName(), TAG_binary))
        {
          // Found the <binary> tag
          has_binary_tag = true;

          // Skip any empty <binary></binary> tags
          if (!currentNode->hasChildNodes())
          {
            continue;
          }

          // Valid mzML does not have any other child nodes except text
          if (currentNode->getChildNodes()->getLength() != 1)
          {
            throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
                "", "Invalid XML: 'binary' element can only have a single, text node child element.");
          }

          // Now we know that the <binary> node has exactly one single child node, the text that we want!
          xercesc::DOMNode* textNode_ = currentNode->getFirstChild();

          if (textNode_->getNodeType() == xercesc::DOMNode::TEXT_NODE)
          {
            xercesc::DOMText* textNode (static_cast<xercesc::DOMText*> (textNode_));
            sm.appendASCII(textNode->getData(),
                textNode->getLength(), data.back().base64);
          }
          else
          {
            throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
                "", "Invalid XML: 'binary' element can only have a single, text node child element.");
          }
        }
        else if (xercesc::XMLString::equals(currentElement->getTagName(), TAG_CV))
        {
          std::string accession = sm.convert(currentElement->getAttribute(TAG_accession));
          std::string value = sm.convert(currentElement->getAttribute(TAG_value));
          std::string name = sm.convert(currentElement->getAttribute(TAG_name));
          std::string unit_accession = sm.convert(currentElement->getAttribute(TAG_unit_accession));

          // set precision, data_type
          Internal::MzMLHandlerHelper::handleBinaryDataArrayCVParam(data, accession, value, name, unit_accession);
        }
        else if (xercesc::XMLString::equals(currentElement->getTagName(), TAG_userParam))
        {
          std::cout << " unhandled userParam" << std::endl;
        }
        else if (xercesc::XMLString::equals(currentElement->getTagName(), TAG_referenceableParamGroupRef))
        {
          std::cout << " unhandled referenceableParamGroupRef" << std::endl;
        }
        else
        {
          //std::cout << "unhandled" << (string)xercesc::XMLString::transcode(currentNode->getNodeName() << std::endl;
        }
      }
    }

    // Throw exception upon invalid mzML: the <binary> tag is required inside <binaryDataArray>
    if (!has_binary_tag)
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
          "", "Invalid XML: 'binary' element needs to be present at least once inside 'binaryDataArray' element.");
    }
  }

  std::string MzMLSpectrumDecoder::domParseString_(const std::string& in, std::vector<BinaryData>& data)
  {
    // The public method name is retained; Expat streams the bounded fragment
    // directly into BinaryData rather than constructing a temporary DOM tree.
    struct Fragment
    {
      XML_Parser parser;
      std::vector<BinaryData>& data;
      std::exception_ptr exception;
      std::string id;
      Size depth = 0, array_depth = 0, binary_depth = 0;
      int length = 0;
      bool binary_seen = false, root_complete = false;

      static const char* attribute(const XML_Char** attributes, const char* name)
      {
        for (Size i = 0; attributes[i]; i += 2)
        {
          if (std::strcmp(attributes[i], name) == 0) return attributes[i + 1];
        }
        return "";
      }
      void capture() noexcept
      {
        if (!exception) exception = std::current_exception();
        XML_StopParser(parser, XML_FALSE);
      }
      [[noreturn]] static void invalid(const char* message)
      {
        throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", message);
      }
      static void XMLCALL start(void* opaque, const XML_Char* name, const XML_Char** attributes) noexcept
      {
        auto& s = *static_cast<Fragment*>(opaque);
        if (s.exception) return;
        try
        {
          ++s.depth;
          const std::string_view tag(name);
          if (s.depth == 1)
          {
            if (tag != "spectrum" && tag != "chromatogram") invalid("Expected spectrum or chromatogram root element");
            const char* size = attribute(attributes, "defaultArrayLength");
            if (!*size) invalid("Root element does not contain defaultArrayLength XML tag");
            s.length = StringUtils::toInt32(size);
            s.id = attribute(attributes, "id");
          }
          if (s.binary_depth) invalid("Invalid XML: binary must contain only text");
          if (tag == "binaryDataArray")
          {
            if (s.array_depth) invalid("Nested binaryDataArray element");
            s.array_depth = s.depth;
            s.binary_seen = false;
            s.data.emplace_back();
            s.data.back().size = s.length;
          }
          else if (s.array_depth && s.depth == s.array_depth + 1)
          {
            if (tag == "binary") { s.binary_depth = s.depth; s.binary_seen = true; }
            else if (tag == "cvParam")
            {
              Internal::MzMLHandlerHelper::handleBinaryDataArrayCVParam(s.data,
                attribute(attributes, "accession"), attribute(attributes, "value"),
                attribute(attributes, "name"), attribute(attributes, "unitAccession"));
            }
            else if (tag == "userParam") { std::cout << " unhandled userParam\n"; }
            else if (tag == "referenceableParamGroupRef") { std::cout << " unhandled referenceableParamGroupRef\n"; }
          }
        }
        catch (...) { s.capture(); }
      }
      static void XMLCALL end(void* opaque, const XML_Char*) noexcept
      {
        auto& s = *static_cast<Fragment*>(opaque);
        if (s.exception) return;
        try
        {
          if (s.depth == s.binary_depth) s.binary_depth = 0;
          if (s.depth == s.array_depth)
          {
            if (!s.binary_seen) invalid("Invalid XML: binary element missing inside binaryDataArray");
            s.array_depth = 0;
          }
          --s.depth;
          if (s.depth == 0)
          {
            // Indexed offsets may include the parent's closing tags after the
            // requested subtree; stop only after its complete root element.
            s.root_complete = true;
            XML_StopParser(s.parser, XML_FALSE);
          }
        }
        catch (...) { s.capture(); }
      }
      static void XMLCALL text(void* opaque, const XML_Char* value, int length) noexcept
      {
        auto& s = *static_cast<Fragment*>(opaque);
        if (s.exception) return;
        try { if (s.binary_depth) s.data.back().base64.append(value, static_cast<Size>(length)); }
        catch (...) { s.capture(); }
      }
      static void XMLCALL nontext(void* opaque) noexcept
      {
        auto& s = *static_cast<Fragment*>(opaque);
        if (s.exception) return;
        try { if (s.binary_depth) invalid("Invalid XML: binary must contain only text"); }
        catch (...) { s.capture(); }
      }
    };
    std::unique_ptr<XML_ParserStruct, decltype(&XML_ParserFree)> parser(XML_ParserCreate(nullptr), XML_ParserFree);
    if (!parser) throw std::bad_alloc();
    Fragment fragment{parser.get(), data, {}, {}};
    XML_SetUserData(parser.get(), &fragment);
    XML_SetElementHandler(parser.get(), Fragment::start, Fragment::end);
    XML_SetCharacterDataHandler(parser.get(), Fragment::text);
    XML_SetStartCdataSectionHandler(parser.get(), Fragment::nontext);
    XML_SetCommentHandler(parser.get(), [](void* state, const XML_Char*) { Fragment::nontext(state); });
    XML_SetProcessingInstructionHandler(parser.get(), [](void* state, const XML_Char*, const XML_Char*) { Fragment::nontext(state); });
    XML_SetParamEntityParsing(parser.get(), XML_PARAM_ENTITY_PARSING_NEVER);
    XML_SetExternalEntityRefHandler(parser.get(), [](XML_Parser, const XML_Char*, const XML_Char*, const XML_Char*, const XML_Char*) -> int { return XML_STATUS_ERROR; });
    for (Size offset = 0;;)
    {
      const auto count = std::min<Size>(65536, in.size() - offset);
      const bool final = offset + count == in.size();
      const auto status = XML_Parse(parser.get(), in.data() + offset, static_cast<int>(count), final);
      if (fragment.exception) std::rethrow_exception(fragment.exception);
      if (status == XML_STATUS_ERROR && !fragment.root_complete) Fragment::invalid(XML_ErrorString(XML_GetErrorCode(parser.get())));
      if (fragment.root_complete || final) break;
      offset += count;
    }
    return fragment.id;
  }

      // depending on the system).
      //

        // TESTING CAST: res = (int) res; // use this only when emulating 32 bit systems

        std::cerr << "Trying to convert corrupted / unreadable value to std::streampos : " << s << std::endl;
        std::cerr << "This can also happen if the value exceeds 63 bits, please check your input." << std::endl;
        throw Exception::ConversionError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            std::string("Could not convert string '") + s + "' to a 64 bit integer.");
      }

      // Check if the value can fit into std::streampos
      if ( std::abs( boost::lexical_cast< long double >(s) - res) > 0.1)
      {
        std::cerr << "Your system may not support addressing a file of this size,"
          << " only addresses that fit into a " << sizeof(std::streamsize)*8 <<
          " bit integer are supported on your system." << std::endl;
        throw Exception::ConversionError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            std::string("Could not convert string '") + s + "' to an integer on your system.");
      }

      return res;
    }
  }

  int IndexedMzMLDecoder::parseOffsets(const std::string& filename, std::streampos indexoffset, OffsetVector& spectra_offsets, OffsetVector& chromatograms_offsets)
  {
    //-------------------------------------------------------------
    // Open file, jump to end and read last indexoffset bytes into buffer.
    //-------------------------------------------------------------
    std::ifstream f(filename.c_str());
    if (!f.is_open())
    {
      if (!File::exists(filename))
      {
        throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else if (!File::readable(filename))
      {
        throw Exception::FileNotReadable(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else
      {
        throw Exception::IOException(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
    }

    // get length of file:
    f.seekg(0, f.end);
    std::streampos length = f.tellg();

    if (indexoffset < 0 || indexoffset > length)
    {
      std::cerr << "IndexedMzMLDecoder::parseOffsets Error: Offset was " <<
        indexoffset << " (not between 0 and " << length << ")." << std::endl;
      return -1;
    }

    //-------------------------------------------------------------
    // Read full end of file to parse offsets for spectra and chroms
    //-------------------------------------------------------------
    // read data as a block into a buffer

    // allocate enough memory in buffer (+1 for string termination)
    std::streampos readl = length - indexoffset;
    std::unique_ptr<char[]> buffer(new(std::nothrow) char[readl + std::streampos(1)]);

    // catch case where not enough memory is available
    if (buffer == nullptr)
    {
      // Warning: Index takes up more than 10 % of the whole file, please check your input file.\n";
      std::cerr << "IndexedMzMLDecoder::parseOffsets Could not allocate enough memory to read in index of indexedMzML" << std::endl;
      std::cerr << "IndexedMzMLDecoder::parseOffsets calculated index offset " << indexoffset << " and file length " << length <<
        ", consequently tried to read into memory " << readl << " bytes." << std::endl;
      return -1;
    }

    // read into memory
    f.seekg(-readl, f.end);
    f.read(buffer.get(), readl);
    buffer[readl] = '\0';

    //-------------------------------------------------------------
    // Add a sane start element and then give it to a DOM parser
    //-------------------------------------------------------------
    // http://stackoverflow.com/questions/4691039/making-xerces-parse-a-string-insted-of-a-file
    std::string tmp_fixed_xml = "<indexedmzML>" +  std::string(buffer.get()) + "\n";
    int res = domParseIndexedEnd_(tmp_fixed_xml, spectra_offsets, chromatograms_offsets);


    return res;
  }

  std::streampos IndexedMzMLDecoder::findIndexListOffset(const std::string& filename, int buffersize)
  {
    // return value
    std::streampos indexoffset = -1;

    //-------------------------------------------------------------
    // Open file, jump to end and read last n bytes into buffer.
    //-------------------------------------------------------------
    std::ifstream f(filename.c_str());

    if (!f.is_open())
    {
      if (!File::exists(filename))
      {
        throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else if (!File::readable(filename))
      {
        throw Exception::FileNotReadable(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else
      {
        throw Exception::IOException(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
    }

    // Read the last few bytes and hope our offset is there to be found
    std::unique_ptr<char[]> buffer(new char[buffersize + 1]);
    f.seekg(-buffersize, f.end);
    f.read(buffer.get(), buffersize);
    buffer.get()[buffersize] = '\0';

#ifdef DEBUG_READER
    std::cout << " reading file " << filename  << " with size " << buffersize << '\n';
    std::cout << buffer << '\n';
#endif

    //-------------------------------------------------------------
    // Since we could be anywhere in the XML structure, use regex to find
    // indexListOffset and read its content.
    //-------------------------------------------------------------
    boost::regex listoffset_rx(R"(<[^>/]*indexListOffset\s*>\s*(\d*))");
    boost::cmatch matches;
    boost::regex_search(buffer.get(), matches, listoffset_rx);
    std::string thismatch(matches[1].first, matches[1].second);
    if (!thismatch.empty())
    {
      try
      {
        indexoffset = OpenMS::IndexedMzMLUtils::stringToStreampos(thismatch);
      }
      catch (Exception::ConversionError& /*e*/)
      {
        std::cerr << "Corrupted / unreadable value in <indexListOffset> : " << thismatch << std::endl;
        // free resources and re-throw
        throw;  // re-throw conversion error
      }
    }
    else
    {
      std::cerr << "IndexedMzMLDecoder::findIndexListOffset Error: Could not find element indexListOffset in the last "
                << buffersize << " bytes. Maybe this is not a indexedMzML."
                << buffer.get() << std::endl;
    }

    return indexoffset;
  }

  int IndexedMzMLDecoder::domParseIndexedEnd_(const std::string& in, OffsetVector& spectra_offsets, OffsetVector& chromatograms_offsets)
  {
    struct Footer
    {
      XML_Parser parser;
      std::exception_ptr exception;
      OffsetVector spectra, chromatograms, entries;
      std::string index_name, id, offset;
      Size depth = 0, list_depth = 0, index_depth = 0, offset_depth = 0, lists = 0;
      bool invalid_index = false, has_spectra = false, has_chromatograms = false;
      static const char* attribute(const XML_Char** attributes, const char* name)
      {
        for (Size i = 0; attributes[i]; i += 2)
        {
          if (std::strcmp(attributes[i], name) == 0) return attributes[i + 1];
        }
        return "";
      }
      void capture() noexcept
      {
        if (!exception) exception = std::current_exception();
        XML_StopParser(parser, XML_FALSE);
      }
      static void XMLCALL start(void* opaque, const XML_Char* name, const XML_Char** attributes) noexcept
      {
        auto& s = *static_cast<Footer*>(opaque);
        if (s.exception) return;
        try
        {
          ++s.depth;
          if (std::string_view(name) == "indexList" && s.depth > 1)
          {
            ++s.lists;
            s.list_depth = s.depth;
          }
          else if (s.list_depth && s.depth == s.list_depth + 1)
          {
            s.index_depth = s.depth;
            s.index_name = attribute(attributes, "name");
            s.entries.clear();
          }
          else if (s.index_depth && s.depth == s.index_depth + 1)
          {
            s.offset_depth = s.depth;
            s.id = attribute(attributes, "idRef");
            s.offset.clear();
          }
        }
        catch (...) { s.capture(); }
      }
      static void XMLCALL end(void* opaque, const XML_Char*) noexcept
      {
        auto& s = *static_cast<Footer*>(opaque);
        if (s.exception) return;
        try
        {
          if (s.depth == s.offset_depth)
          {
            s.entries.emplace_back(s.id, IndexedMzMLUtils::stringToStreampos(s.offset));
            s.offset_depth = 0;
          }
          if (s.depth == s.index_depth)
          {
            if (s.index_name == "spectrum") { s.spectra = std::move(s.entries); s.has_spectra = true; }
            else if (s.index_name == "chromatogram") { s.chromatograms = std::move(s.entries); s.has_chromatograms = true; }
            else { s.invalid_index = true; }
            s.index_depth = 0;
          }
          if (s.depth == s.list_depth) s.list_depth = 0;
          --s.depth;
        }
        catch (...) { s.capture(); }
      }
      static void XMLCALL text(void* opaque, const XML_Char* value, int length) noexcept
      {
        auto& s = *static_cast<Footer*>(opaque);
        if (s.exception) return;
        try { if (s.offset_depth) s.offset.append(value, static_cast<Size>(length)); }
        catch (...) { s.capture(); }
      }
    };
    std::unique_ptr<XML_ParserStruct, decltype(&XML_ParserFree)> parser(XML_ParserCreate(nullptr), XML_ParserFree);
    if (!parser) throw std::bad_alloc();
    Footer footer{};
    footer.parser = parser.get();
    XML_SetUserData(parser.get(), &footer);
    XML_SetElementHandler(parser.get(), Footer::start, Footer::end);
    XML_SetCharacterDataHandler(parser.get(), Footer::text);
    XML_SetParamEntityParsing(parser.get(), XML_PARAM_ENTITY_PARSING_NEVER);
    XML_SetExternalEntityRefHandler(parser.get(), [](XML_Parser, const XML_Char*, const XML_Char*, const XML_Char*, const XML_Char*) -> int { return XML_STATUS_ERROR; });
    for (Size offset = 0;;)
    {
      const auto count = std::min<Size>(65536, in.size() - offset);
      const bool final = offset + count == in.size();
      const auto status = XML_Parse(parser.get(), in.data() + offset, static_cast<int>(count), final);
      if (footer.exception) std::rethrow_exception(footer.exception);
      if (status == XML_STATUS_ERROR)
      {
        throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", XML_ErrorString(XML_GetErrorCode(parser.get())));
      }
      if (final) break;
      offset += count;
    }
    if (footer.lists != 1 || footer.invalid_index) return -1;
    if (footer.has_spectra) spectra_offsets = std::move(footer.spectra);
    if (footer.has_chromatograms) chromatograms_offsets = std::move(footer.chromatograms);
    return 0;
  }

}

  {

    {
      const auto i = index(qname);
      return i < 0 ? nullptr : pairs_[2 * i + 1];
    }

  {
    if (pairs_)
    {
      // Expat names are UTF-8 regardless of the process locale.
      const xercesc::TranscodeFromStr name(reinterpret_cast<const XMLByte*>(qname), std::strlen(qname), "UTF-8");
      return value(reinterpret_cast<const char16_t*>(name.str()));
    }
    ::XMLCh* transcoded = xercesc::XMLString::transcode(qname);
    const char16_t* val = value(reinterpret_cast<const char16_t*>(transcoded));

    return val;

      return -1;
    }
    return static_cast<SignedSize>(attrs(attributes_)->getIndex(reinterpret_cast<const ::XMLCh*>(qname)));
  }

      }
      return nullptr;
    }
    return reinterpret_cast<const char16_t*>(attrs(attributes_)->getValue(static_cast<XMLSize_t>(i)));

    void parse(xercesc::InputSource* const source, XMLHandler* handler)
    {
      // Deliberately DOM-backed event delivery: this prototype measures the whole-file
      // memory cost of pugixml rather than claiming to supply streaming parsing.
      // ponytail: the DOM is O(file size); use a streaming backend for bounded-memory loads.
      auto utf16 = [](const char* value)
      {
        const auto wide = pugi::as_wide(value);
        std::u16string result;
        for (auto ch : wide)
        {
          const auto cp = static_cast<uint32_t>(ch);
          if constexpr (sizeof(wchar_t) == 2) result.push_back(static_cast<char16_t>(cp));
          else if (cp <= 0xffff) result.push_back(static_cast<char16_t>(cp));
          else
          {
            result.push_back(static_cast<char16_t>(0xd800 + ((cp - 0x10000) >> 10)));
            result.push_back(static_cast<char16_t>(0xdc00 + ((cp - 0x10000) & 0x3ff)));
          }
        }
        return result;
      };
      unique_ptr<xercesc::BinInputStream> stream(source->makeStream());
      if (!stream)
      {
        throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", "Cannot open XML input stream");
      }
      std::vector<char> input;
      std::array<XMLByte, 65536> chunk;
      XMLSize_t count;
      while ((count = stream->readBytes(chunk.data(), chunk.size())) != 0)
      {
        input.insert(input.end(), chunk.begin(), chunk.begin() + count);
      }
      pugi::xml_document document;
      // Keep whitespace text and CDATA so handler character data is preserved.
      pugi::xml_encoding encoding = pugi::encoding_auto;
      if (source->getEncoding())
      {
        const std::string requested = StringManager::convert(reinterpret_cast<const char16_t*>(source->getEncoding()));
        if (requested == "UTF-8" || requested == "utf-8") encoding = pugi::encoding_utf8;
        else if (requested == "UTF-16" || requested == "utf-16") encoding = pugi::encoding_utf16;
        else if (requested == "ISO-8859-1") encoding = pugi::encoding_latin1;
        else
        {
          throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", "Unsupported pugixml encoding override: " + requested);
        }
      }
      const auto status = document.load_buffer_inplace(input.data(), input.size(), pugi::parse_default | pugi::parse_ws_pcdata, encoding);
      if (!status)
      {
        throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "",
            std::string("pugixml: ") + status.description() + " at byte " + std::to_string(status.offset));
      }
      std::function<void(pugi::xml_node)> visit = [&](pugi::xml_node node)
      {
        if (node.type() == pugi::node_element)
        {
          auto name = utf16(node.name());
          XMLAttributes::Values attributes;
          for (auto attr : node.attributes()) attributes.emplace_back(utf16(attr.name()), utf16(attr.value()));
          handler->onStartElement(name.c_str(), XMLAttributes(attributes));
          for (auto child : node.children()) visit(child);
          handler->onEndElement(name.c_str());
        }
        else if (node.type() == pugi::node_pcdata || node.type() == pugi::node_cdata)
        {
          auto text = utf16(node.value());
          handler->onCharacters(text.c_str(), text.size());
        }
      };
      try
      {
        for (auto child : document.children()) visit(child);
      }
      catch (const XMLHandler::EndParsingSoftly&)
      {
        // Early callback termination works, but the DOM has already been built.
      }
    }

    void XMLFile::parse_(const std::string & filename, XMLHandler * handler)
    {
      // ensure handler->reset() is called to save memory (in case the XMLFile
      // reader, e.g. FeatureXMLFile, is used again)
      XMLCleaner_ clean(handler);

      StringManager sm;
      //try to open file
      if (!File::exists(filename))
      {
        throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }

      // initialize parser
      try
      {
        Internal::xercesInitialize(); // serialised: Xerces requires it, and parseBuffer_() runs on several threads at once
      }
      catch (const xercesc::XMLException & toCatch)
      {
        throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            "",std::string("Error during initialization: ") + StringManager().convert(toCatch.getMessage()));
      }


      // peak ahead into the file: is it bzip2 or gzip compressed?
      std::string bz;
      {
        std::ifstream file(filename.c_str());
        char tmp_bz[3];
        file.read(tmp_bz, 2);
        tmp_bz[2] = '\0';
        bz =std::string(tmp_bz);
      }

      unique_ptr<xercesc::InputSource> source;

      char g1 = 0x1f;
      char g2 = 0;
      g2 |= 1 << 7;
      g2 |= 1 << 3;
      g2 |= 1 << 1;
      g2 |= 1 << 0;
      //g2 = static_cast<char>(0x8b); // can make troubles if it is casted to 0x7F which is the biggest number signed char can save
      if ((bz[0] == 'B' && bz[1] == 'Z') || (bz[0] == g1 && bz[1] == g2) || (bz[0] == 'P' && bz[1] == 'K'))
      {
        source.reset(new CompressedInputSource(sm.convert(filename).c_str(), bz));
      }
      else
      {
        source.reset(new xercesc::LocalFileInputSource(sm.convert(filename).c_str()));
      }
      // what if no encoding given http://xerces.apache.org/xerces-c/apiDocs-3/classInputSource.html
      if (!enforced_encoding_.empty())
      {
        static const XMLCh* s_enc = xercesc::XMLString::transcode(enforced_encoding_.c_str());
        source->setEncoding(s_enc);
      }
      
      parse(source.get(), handler);
    }

    void XMLFile::save_(const std::string & filename, XMLHandler * handler) const
    {
      // Detect compression from filename extension, in any letter case like on input (and in TOPPBase's check)
      const FileTypes::Type compression = FileNameUtils::compressionType(filename);
      const bool use_gzip = compression == FileTypes::GZ;
      const bool use_bzip2 = compression == FileTypes::BZ2;

      if (use_gzip)
      {
        // Minimal std::streambuf that writes gzip-compressed data via zlib
        class GzipOStreambuf : public std::streambuf
        {
        public:
          enum { BUF_SIZE = 65536 };

          explicit GzipOStreambuf(const char* fn) : gz_(gzopen(fn, "wb"))
          {
            if (gz_) setp(buf_, buf_ + BUF_SIZE);
          }

          ~GzipOStreambuf()
          {
            flush_buffer();
            if (gz_) gzclose(gz_);
          }

          bool is_open() const { return gz_ != nullptr; }

        protected:
          int_type overflow(int_type c) override
          {
            if (!gz_) return traits_type::eof();
            if (pptr() == epptr() && !flush_buffer()) return traits_type::eof();
            if (!traits_type::eq_int_type(c, traits_type::eof()))
            {
              *pptr() = traits_type::to_char_type(c);
              pbump(1);
            }
            return traits_type::not_eof(c);
          }

          int sync() override { return flush_buffer() ? 0 : -1; }

        private:
          bool flush_buffer()
          {
            int n = static_cast<int>(pptr() - pbase());
            if (n > 0)
            {
              if (gzwrite(gz_, pbase(), static_cast<unsigned>(n)) != n) return false;
              setp(buf_, buf_ + BUF_SIZE);
            }
            return true;
          }

          gzFile gz_;
          char buf_[BUF_SIZE];
        };

        GzipOStreambuf sbuf(filename.c_str());
        if (!sbuf.is_open())
        {
          throw Exception::UnableToCreateFile(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
        }
        std::ostream os(&sbuf);
        os.precision(writtenDigits(double()));
        handler->writeTo(os);
      }
      else if (use_bzip2)
      {
        // Minimal std::streambuf that writes bzip2-compressed data via bzlib
        class Bzip2OStreambuf : public std::streambuf
        {
        public:
          enum { BUF_SIZE = 65536 };

          explicit Bzip2OStreambuf(const char* fn) : file_(nullptr), bz_(nullptr)
          {
            file_ = fopen(fn, "wb");
            if (!file_) return;
            int err = BZ_OK;
            bz_ = BZ2_bzWriteOpen(&err, file_, 9, 0, 0);
            if (err != BZ_OK)
            {
              BZ2_bzWriteClose(&err, bz_, 1, nullptr, nullptr);
              bz_ = nullptr;
              fclose(file_);
              file_ = nullptr;
              return;
            }
            setp(buf_, buf_ + BUF_SIZE);
          }

          ~Bzip2OStreambuf()
          {
            flush_buffer();
            if (bz_)
            {
              int err = BZ_OK;
              BZ2_bzWriteClose(&err, bz_, 0, nullptr, nullptr);
            }
            if (file_) fclose(file_);
          }

          bool is_open() const { return bz_ != nullptr; }

        protected:
          int_type overflow(int_type c) override
          {
            if (!bz_) return traits_type::eof();
            if (pptr() == epptr() && !flush_buffer()) return traits_type::eof();
            if (!traits_type::eq_int_type(c, traits_type::eof()))
            {
              *pptr() = traits_type::to_char_type(c);
              pbump(1);
            }
            return traits_type::not_eof(c);
          }

          int sync() override { return flush_buffer() ? 0 : -1; }

        private:
          bool flush_buffer()
          {
            if (!bz_) return false;
            int n = static_cast<int>(pptr() - pbase());
            if (n > 0)
            {
              int err = BZ_OK;
              BZ2_bzWrite(&err, bz_, pbase(), n);
              if (err != BZ_OK && err != BZ_RUN_OK) return false;
              setp(buf_, buf_ + BUF_SIZE);
            }
            return true;
          }

          FILE* file_;
          BZFILE* bz_;
          char buf_[BUF_SIZE];
        };

        Bzip2OStreambuf sbuf(filename.c_str());
        if (!sbuf.is_open())
        {
          throw Exception::UnableToCreateFile(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
        }
        std::ostream os(&sbuf);
        os.precision(writtenDigits(double()));
        handler->writeTo(os);
      }
      else
      {
        // Uncompressed: open in binary mode to avoid any line ending conversions
        std::ofstream os(filename.c_str(), std::ios::out | std::ios::binary);
        os.precision(writtenDigits(double()));
        if (!os)
        {
          throw Exception::UnableToCreateFile(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
        }
        handler->writeTo(os);
        os.close();
      }
    }

    std::string encodeTab(const std::string& to_encode)
    {
      if (!StringUtils::has(to_encode, '\t'))
      {
        return to_encode;
      }
      else
      {
        std::string result = to_encode;
        StringUtils::substitute(result, "\t", "&#x9;");
        return result;
      }
    }

    bool XMLFile::isValid(const std::string & filename, std::ostream & os)
    {
      if (schema_location_.empty())
      {
        throw Exception::NotImplemented(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION);
      }
      std::string current_location = File::find(schema_location_);
      return XMLValidator().isValid(filename, current_location, os);
    }

    const std::string & XMLFile::getVersion() const
    {
      return schema_version_;
    }

  /// Small internal function to check the default data vectors
  void checkData_(std::vector<Internal::MzMLHandlerHelper::BinaryData>& data,
      SignedSize x_index, SignedSize int_index,
      bool x_precision_64, bool int_precision_64)
  {
    // Error if intensity or m/z (RT) is encoded as int32|64 - they should be float32|64!
    if ((!data[x_index].ints_32.empty()) || (!data[x_index].ints_64.empty()))
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
          "", "Encoding m/z or RT array as integer is not allowed!");
    }
    if ((!data[int_index].ints_32.empty()) || (!data[int_index].ints_64.empty()))
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
          "", "Encoding intensity array as integer is not allowed!");
    }

    Size mz_size = x_precision_64 ? data[x_index].floats_64.size() : data[x_index].floats_32.size();
    Size int_size = int_precision_64 ? data[int_index].floats_64.size() : data[int_index].floats_32.size();

    // Check if int-size and mz-size are equal
    if (mz_size != int_size)
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
          "", "Error, intensity and m/z array length are unequal");
    }
  }

  inline void fillDataArray(const std::vector<Internal::MzMLHandlerHelper::BinaryData>& data,
                            const OpenMS::Interfaces::BinaryDataArrayPtr& array, bool precision_64, SignedSize index)
  {
    // This seems to be the fastest method to move the data (faster than copy or assign)
    if (precision_64)
    {
      array->data.insert(array->data.begin(), data[index].floats_64.begin(), data[index].floats_64.end());
    }
    else
    {
      array->data.insert(array->data.begin(), data[index].floats_32.begin(), data[index].floats_32.end());
    }
  }

  template<class T>
  inline void fillDataArrayFloat(const Internal::MzMLHandlerHelper::BinaryData& data, T& spectrum)
  {
    //create new array
    spectrum.getFloatDataArrays().resize(spectrum.getFloatDataArrays().size() + 1);
    //reserve space in the array
    spectrum.getFloatDataArrays().back().reserve(data.size);
    //copy meta info into MetaInfoDescription
    spectrum.getFloatDataArrays().back().MetaInfoDescription::operator=(data.meta);

    if (data.precision == Internal::MzMLHandlerHelper::BinaryData::PRE_64)
    {
      for (Size n = 0; n < data.floats_64.size(); n++)
      {
        double value = data.floats_64[n];
        spectrum.getFloatDataArrays().back().push_back(value);
      }
    }
    else
    {
      for (Size n = 0; n < data.floats_32.size(); n++)
      {
        double value = data.floats_32[n];
        spectrum.getFloatDataArrays().back().push_back(value);
      }
    }
  }

  inline void fillDataArrays(const std::vector<Internal::MzMLHandlerHelper::BinaryData>& input_data, MSSpectrum& spectrum)
  {
    for (Size i = 0; i < input_data.size(); i++) //loop over all binary data arrays
    {
      if (input_data[i].meta.getName() != "m/z array" && input_data[i].meta.getName() != "intensity array") // is meta data array?
      {
        if (input_data[i].data_type == Internal::MzMLHandlerHelper::BinaryData::DT_FLOAT)
        {
          fillDataArrayFloat(input_data[i], spectrum);
        }
        else if (input_data[i].data_type == Internal::MzMLHandlerHelper::BinaryData::DT_INT)
        {
          fillDataArrayInt(input_data[i], spectrum);
        }
        else if (input_data[i].data_type == Internal::MzMLHandlerHelper::BinaryData::DT_STRING)
        {
          fillDataArrayString(input_data[i], spectrum);
        }
      }
    }
  }

  template<class T>
  inline void fillDataArray(const std::vector<Internal::MzMLHandlerHelper::BinaryData>& data,
                            T& peak_container,
                            const bool x_precision_64,
                            const bool int_precision_64,
                            const SignedSize x_index,
                            const SignedSize int_index,
                            const Size default_array_length_)
  {
    typename T::PeakType tmp;
    if (x_precision_64 && !int_precision_64)
    {
      std::vector< double >::const_iterator mz_it = data[x_index].floats_64.begin();
      std::vector< float >::const_iterator int_it = data[int_index].floats_32.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
    else if (x_precision_64 && int_precision_64)
    {
      std::vector< double >::const_iterator mz_it = data[x_index].floats_64.begin();
      std::vector< double >::const_iterator int_it = data[int_index].floats_64.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
    else if (!x_precision_64 && int_precision_64)
    {
      std::vector< float >::const_iterator mz_it = data[x_index].floats_32.begin();
      std::vector< double >::const_iterator int_it = data[int_index].floats_64.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
    else if (!x_precision_64 && !int_precision_64)
    {
      std::vector< float >::const_iterator mz_it = data[x_index].floats_32.begin();
      std::vector< float >::const_iterator int_it = data[int_index].floats_32.begin();
      for (Size n = 0; n < default_array_length_; n++)
      {
        //add peak
        tmp.setIntensity(*int_it);
        tmp.setPos(*mz_it);
        ++mz_it;
        ++int_it;
        peak_container.push_back(tmp);
      }
    }
  }

  void MzMLSpectrumDecoder::decodeBinaryDataMSSpectrum_(std::vector<BinaryData>& data, OpenMS::MSSpectrum& spectrum) const
  {
    Internal::MzMLHandlerHelper::decodeBase64Arrays(data, skip_xml_checks_);

    //look up the precision and the index of the intensity and m/z array
    bool x_precision_64 = true;
    bool int_precision_64 = true;
    SignedSize x_index = -1;
    SignedSize int_index = -1;
    Internal::MzMLHandlerHelper::computeDataProperties_(data, x_precision_64, x_index, "m/z array");
    Internal::MzMLHandlerHelper::computeDataProperties_(data, int_precision_64, int_index, "intensity array");

    //Abort if no m/z or intensity array is present
    if (int_index == -1 || x_index == -1)
    {
      std::cerr << "Error, intensity or m/z array is missing, skipping this spectrum" << std::endl;
      return;
    }

    checkData_(data, x_index, int_index, x_precision_64, int_precision_64);

    // At this point, we could check whether the defaultArrayLength from the
    // spectrum/chromatogram tag is equivalent to size of the decoded data
    Size default_array_length_ = x_precision_64 ? data[x_index].floats_64.size() : data[x_index].floats_32.size();

    spectrum.reserve(default_array_length_);
    fillDataArray(data, spectrum,
                  x_precision_64, int_precision_64,
                  x_index, int_index, default_array_length_);

    if (data.size() > 2) 
    {
      fillDataArrays(data, spectrum);
    }
  }

  void MzMLSpectrumDecoder::handleBinaryDataArray_(void* indexListNode_opaque, std::vector<BinaryData>& data)
  {
    auto node = *static_cast<pugi::xml_node*>(indexListNode_opaque);
    data.emplace_back();
    bool has_binary = false;
    for (auto child : node.children())
    {
      if (child.type() != pugi::node_element) continue;
      const std::string name = child.name();
      if (name == "binary")
      {
        has_binary = true;
        const auto text = child.first_child();
        if (!text) continue;
        if (text.type() != pugi::node_pcdata || text.next_sibling())
        {
          throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
              "", "Invalid XML: 'binary' element can only have a single, text node child element.");
        }
        data.back().base64 = text.value();
      }
      else if (name == "cvParam")
      {
        Internal::MzMLHandlerHelper::handleBinaryDataArrayCVParam(data,
            child.attribute("accession").value(), child.attribute("value").value(),
            child.attribute("name").value(), child.attribute("unitAccession").value());
      }
      else if (name == "userParam" || name == "referenceableParamGroupRef")
      {
        std::cout << " unhandled " << name << std::endl;
      }
    }
    if (!has_binary)
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
          "", "Invalid XML: 'binary' element needs to be present at least once inside 'binaryDataArray' element.");
    }
  }

  std::string MzMLSpectrumDecoder::domParseString_(const std::string& in, std::vector<BinaryData>& data)
  {
    // Only the requested spectrum/chromatogram fragment is materialized, never the whole mzML.
    // Indexed readers delimit records by the next record offset. The last record
    // can therefore be followed by ancestor closing tags. Frame exactly one root,
    // then let pugixml strictly parse that root (never recover malformed contents).
    Size fragment_end = in.size();
    Size depth = 0;
    for (Size pos = 0; pos < in.size();)
    {
      pos = in.find('<', pos);
      if (pos == std::string::npos) break;
      const char* terminator = nullptr;
      Size prefix = 0;
      if (in.compare(pos, 4, "<!--") == 0) { terminator = "-->"; prefix = 4; }
      else if (in.compare(pos, 9, "<![CDATA[") == 0) { terminator = "]]>"; prefix = 9; }
      else if (in.compare(pos, 2, "<?") == 0) { terminator = "?>"; prefix = 2; }
      if (terminator)
      {
        const auto end = in.find(terminator, pos + prefix);
        if (end == std::string::npos) break;
        pos = end + std::char_traits<char>::length(terminator);
        continue;
      }
      const bool closing = pos + 1 < in.size() && in[pos + 1] == '/';
      char quote = 0;
      Size end = pos + 1;
      for (; end < in.size(); ++end)
      {
        const char ch = in[end];
        if (quote) { if (ch == quote) quote = 0; }
        else if (ch == '\'' || ch == '"') quote = ch;
        else if (ch == '>') break;
      }
      if (end == in.size()) break;
      if (closing)
      {
        if (depth == 0) break;
        --depth;
      }
      else if (in[end - 1] != '/') ++depth;
      if (depth == 0)
      {
        fragment_end = end + 1;
        break;
      }
      pos = end + 1;
    }
    pugi::xml_document document;
    const auto status = document.load_buffer(in.data(), fragment_end, pugi::parse_default | pugi::parse_comments);
    if (!status)
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "",
          std::string("pugixml: ") + status.description() + " at byte " + std::to_string(status.offset));
    }
    auto root = document.document_element();
    if (!root || (std::string(root.name()) != "spectrum" && std::string(root.name()) != "chromatogram"))
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", "Expected spectrum or chromatogram root element");
    }
    if (!root.attribute("defaultArrayLength"))
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "", "Root element does not contain defaultArrayLength XML tag.");
    }
    const int array_length = StringUtils::toInt32(root.attribute("defaultArrayLength").value());
    // Match getElementsByTagName traversal without XPath or dependence on namespace declarations.
    std::function<void(pugi::xml_node)> visit = [&](pugi::xml_node node)
    {
      for (auto child : node.children())
      {
        if (std::string(child.name()) == "binaryDataArray")
        {
          handleBinaryDataArray_(&child, data);
          data.back().size = array_length;
        }
        visit(child);
      }
    };
    visit(root);
    return root.attribute("id").value();
  }

  void MzMLSpectrumDecoder::domParseSpectrum(const std::string& in, OpenMS::Interfaces::SpectrumPtr& sptr)
  {
    std::vector<BinaryData> data;
    domParseString_(in, data);

    std::streampos stringToStreampos(const std::string& s)
    {
      // Try to cast the string to a type that can hold the integer value
      // stored in the std::streampos type (which can vary from 32 to 128 bits
      // depending on the system).
      //
      // long long has a minimal size of 64 bits and can address a range up to
      // 16 Exbibit (or 2 Exbibyte), we can hopefully expect our files to be
      // smaller than an Exabyte and should be safe.
      std::streampos res;
      try
      {

        res = boost::lexical_cast< unsigned long long >(s);
        // TESTING CAST: res = (int) res; // use this only when emulating 32 bit systems

      }
      catch (boost::bad_lexical_cast&)
      {
        std::cerr << "Trying to convert corrupted / unreadable value to std::streampos : " << s << std::endl;
        std::cerr << "This can also happen if the value exceeds 63 bits, please check your input." << std::endl;
        throw Exception::ConversionError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            std::string("Could not convert string '") + s + "' to a 64 bit integer.");
      }

      // Check if the value can fit into std::streampos
      if ( std::abs( boost::lexical_cast< long double >(s) - res) > 0.1)
      {
        std::cerr << "Your system may not support addressing a file of this size,"
          << " only addresses that fit into a " << sizeof(std::streamsize)*8 <<
          " bit integer are supported on your system." << std::endl;
        throw Exception::ConversionError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION,
            std::string("Could not convert string '") + s + "' to an integer on your system.");
      }

      return res;
    }

  int IndexedMzMLDecoder::parseOffsets(const std::string& filename, std::streampos indexoffset, OffsetVector& spectra_offsets, OffsetVector& chromatograms_offsets)
  {
    //-------------------------------------------------------------
    // Open file, jump to end and read last indexoffset bytes into buffer.
    //-------------------------------------------------------------
    std::ifstream f(filename.c_str());
    if (!f.is_open())
    {
      if (!File::exists(filename))
      {
        throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else if (!File::readable(filename))
      {
        throw Exception::FileNotReadable(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else
      {
        throw Exception::IOException(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
    }

    // get length of file:
    f.seekg(0, f.end);
    std::streampos length = f.tellg();

    if (indexoffset < 0 || indexoffset > length)
    {
      std::cerr << "IndexedMzMLDecoder::parseOffsets Error: Offset was " <<
        indexoffset << " (not between 0 and " << length << ")." << std::endl;
      return -1;
    }

    //-------------------------------------------------------------
    // Read full end of file to parse offsets for spectra and chroms
    //-------------------------------------------------------------
    // read data as a block into a buffer

    // allocate enough memory in buffer (+1 for string termination)
    std::streampos readl = length - indexoffset;
    char* buffer = new(std::nothrow) char[readl + std::streampos(1)];

    // catch case where not enough memory is available
    if (buffer == nullptr)
    {
      // Warning: Index takes up more than 10 % of the whole file, please check your input file.\n";
      std::cerr << "IndexedMzMLDecoder::parseOffsets Could not allocate enough memory to read in index of indexedMzML" << std::endl;
      std::cerr << "IndexedMzMLDecoder::parseOffsets calculated index offset " << indexoffset << " and file length " << length <<
        ", consequently tried to read into memory " << readl << " bytes." << std::endl;
      return -1;
    }

    // read into memory
    f.seekg(-readl, f.end);
    f.read(buffer, readl);
    buffer[readl] = '\0';

    //-------------------------------------------------------------
    // Add a sane start element and then give it to a DOM parser
    //-------------------------------------------------------------
    // http://stackoverflow.com/questions/4691039/making-xerces-parse-a-string-insted-of-a-file
    std::string tmp_fixed_xml = "<indexedmzML>" +  std::string(buffer) + "\n";
    int res = domParseIndexedEnd_(tmp_fixed_xml, spectra_offsets, chromatograms_offsets);

    delete[] buffer;

    return res;
  }

  std::streampos IndexedMzMLDecoder::findIndexListOffset(const std::string& filename, int buffersize)
  {
    // return value
    std::streampos indexoffset = -1;

    //-------------------------------------------------------------
    // Open file, jump to end and read last n bytes into buffer.
    //-------------------------------------------------------------
    std::ifstream f(filename.c_str());

    if (!f.is_open())
    {
      if (!File::exists(filename))
      {
        throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else if (!File::readable(filename))
      {
        throw Exception::FileNotReadable(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
      else
      {
        throw Exception::IOException(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
      }
    }

    // Read the last few bytes and hope our offset is there to be found
    std::unique_ptr<char[]> buffer(new char[buffersize + 1]);
    f.seekg(-buffersize, f.end);
    f.read(buffer.get(), buffersize);
    buffer.get()[buffersize] = '\0';

#ifdef DEBUG_READER
    std::cout << " reading file " << filename  << " with size " << buffersize << '\n';
    std::cout << buffer << '\n';
#endif

    //-------------------------------------------------------------
    // Since we could be anywhere in the XML structure, use regex to find
    // indexListOffset and read its content.
    //-------------------------------------------------------------
    boost::regex listoffset_rx(R"(<[^>/]*indexListOffset\s*>\s*(\d*))");
    boost::cmatch matches;
    boost::regex_search(buffer.get(), matches, listoffset_rx);
    std::string thismatch(matches[1].first, matches[1].second);
    if (!thismatch.empty())
    {
      try
      {
        indexoffset = OpenMS::IndexedMzMLUtils::stringToStreampos(thismatch);
      }
      catch (Exception::ConversionError& /*e*/)
      {
        std::cerr << "Corrupted / unreadable value in <indexListOffset> : " << thismatch << std::endl;
        // free resources and re-throw
        throw;  // re-throw conversion error
      }
    }
    else
    {
      std::cerr << "IndexedMzMLDecoder::findIndexListOffset Error: Could not find element indexListOffset in the last "
                << buffersize << " bytes. Maybe this is not a indexedMzML."
                << buffer.get() << std::endl;
    }

    return indexoffset;
  }

  int IndexedMzMLDecoder::domParseIndexedEnd_(const std::string& in, OffsetVector& spectra_offsets, OffsetVector& chromatograms_offsets)
  {

    /*

     We parse something like 
     
      <indexedmzML>
        <indexList count="1">
          <index name="chromatogram">
            <offset idRef="1">9752</offset>
          </index>
        </indexList>
        <indexListOffset>26795</indexListOffset>
      <fileChecksum>0</fileChecksum>
      </indexedmzML>

    */

    pugi::xml_document document;
    const auto status = document.load_buffer(in.data(), in.size());
    if (!status)
    {
      throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, "",
          std::string("pugixml: ") + status.description() + " at byte " + std::to_string(status.offset));
    }
    const auto lists = document.select_nodes("//indexList");
    if (lists.size() != 1) return -1;
    for (auto index : lists[0].node().children())
    {
      if (index.type() != pugi::node_element) continue;
      OffsetVector result;
      for (auto offset : index.children())
      {
        if (offset.type() != pugi::node_element) continue;
        result.emplace_back(offset.attribute("idRef").value(),
                            IndexedMzMLUtils::stringToStreampos(offset.child_value()));
      }
      const std::string name = index.attribute("name").value();
      if (name == "spectrum") spectra_offsets = std::move(result);
      else if (name == "chromatogram") chromatograms_offsets = std::move(result);
      else return -1;
    }
    return 0;
  }

  const char16_t* XMLAttributes::value(const char16_t* qname) const
  {
    if (native_)
    {
      for (const auto& item : *static_cast<const Values*>(attributes_))
      {
        if (item.first == qname) return item.second.c_str();
      }
      return nullptr;
    }

  {
    // Narrow OpenMS names are UTF-8, irrespective of the process locale.
    const xercesc::TranscodeFromStr transcoded(reinterpret_cast<const XMLByte*>(qname), std::strlen(qname), "UTF-8");
    return value(reinterpret_cast<const char16_t*>(transcoded.str()));

  class XMLValidatorErrorHandler_
  {
  public:
    struct Context
    {
      XMLValidator* validator;
      std::exception_ptr exception;
    };
    static void error(void* ctx, const xmlError* error) noexcept
    {
      auto& state = *static_cast<Context*>(ctx);
      if (state.exception) { return; }
      try
      {
        state.validator->logError_(error->level == XML_ERR_WARNING ? "warning" : "error",
                                  error->message ? error->message : "libxml2 validation error",
                                  error->line, error->int2);
      }
      catch (...) { state.exception = std::current_exception(); }
    }
  };

  XMLValidator::XMLValidator() : valid_(true), os_(nullptr) {}

  bool XMLValidator::isValid(const std::string& filename, const std::string& schema, std::ostream& os)
  {
    filename_ = filename;
    os_ = &os;
    valid_ = true;
    if (!File::exists(filename))
    {
      throw Exception::FileNotFound(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, filename);
    }
    unique_ptr<xmlSchemaParserCtxt, decltype(&xmlSchemaFreeParserCtxt)> parser(
      xmlSchemaNewParserCtxt(schema.c_str()), xmlSchemaFreeParserCtxt);
    if (!parser) { throw std::bad_alloc(); }
    XMLValidatorErrorHandler_::Context errors{this, {}};
    xmlSchemaSetParserStructuredErrors(parser.get(), XMLValidatorErrorHandler_::error, &errors);
    unique_ptr<xmlSchema, decltype(&xmlSchemaFree)> grammar(xmlSchemaParse(parser.get()), xmlSchemaFree);
    if (errors.exception) { std::rethrow_exception(errors.exception); }
    if (!grammar) { return false; }
    unique_ptr<xmlSchemaValidCtxt, decltype(&xmlSchemaFreeValidCtxt)> validator(
      xmlSchemaNewValidCtxt(grammar.get()), xmlSchemaFreeValidCtxt);
    if (!validator) { throw std::bad_alloc(); }
    xmlSchemaSetValidStructuredErrors(validator.get(), XMLValidatorErrorHandler_::error, &errors);
    // Streaming validation preserves bounded memory for large mzML inputs.
    const int result = xmlSchemaValidateFile(validator.get(), filename.c_str(), XML_PARSE_NONET | XML_PARSE_NO_XXE);
    if (errors.exception) { std::rethrow_exception(errors.exception); }
    return valid_ && result == 0;
  }

  void XMLValidator::logError_(const std::string& kind, const std::string& message, unsigned long line, unsigned long column)
  {
    (*os_) << "Validation " << kind << " in file '" << filename_ << "' line " << line
           << " column " << column << ": " << message << '\n';
    valid_ = false;

  // Hash every decoded coordinate/intensity, independently of XML text formatting.
  struct Digest final : OpenMS::Interfaces::IMSDataConsumer
  {
    std::uint64_t hash = 14695981039346656037ULL;
    std::uint64_t spectra = 0, chromatograms = 0, points = 0;

    void word(std::uint64_t value)
    {
      for (unsigned i = 0; i < 8; ++i)
      {
        hash = (hash ^ ((value >> (8 * i)) & 255)) * 1099511628211ULL;
      }
    }

    void text(const std::string& value)
    {
      word(value.size());
      for (const unsigned char character : value) { hash = (hash ^ character) * 1099511628211ULL; }
    }

    void consumeSpectrum(OpenMS::MSSpectrum& spectrum) override
    {
      ++spectra;
      points += spectrum.size();
      word(1);
      word(spectrum.size());
      text(spectrum.getNativeID());
      for (const auto& peak : spectrum)
      {
        word(std::bit_cast<std::uint64_t>(peak.getMZ()));
        word(std::bit_cast<std::uint64_t>(static_cast<double>(peak.getIntensity())));
      }
    }

    void consumeChromatogram(OpenMS::MSChromatogram& chromatogram) override
    {
      ++chromatograms;
      points += chromatogram.size();
      word(2);
      word(chromatogram.size());
      text(chromatogram.getNativeID());
      for (const auto& peak : chromatogram)
      {
        word(std::bit_cast<std::uint64_t>(peak.getRT()));
        word(std::bit_cast<std::uint64_t>(static_cast<double>(peak.getIntensity())));
      }
    }

    void setExpectedSize(std::size_t, std::size_t) override {}
    void setExperimentalSettings(const OpenMS::ExperimentalSettings&) override {}
    bool operator==(const Digest& other) const
    {
      return hash == other.hash && spectra == other.spectra &&
        chromatograms == other.chromatograms && points == other.points;
    }
  };

  struct Result
  {
    Digest digest;
    // Keep opened contexts alive until every worker has completed.
    std::shared_ptr<OpenMS::OnDiscMSExperiment> retained_context;
    std::shared_ptr<OpenMS::MSExperiment> retained_experiment;
    double open_seconds = 0;
    double read_seconds = 0;
    std::size_t index_spectra = 0, index_chromatograms = 0;
  };

  Result run(const std::string& mode, const std::string& path, std::size_t queries)
  {
    Result result;
    const auto begin = Clock::now();
    if (mode == "stream")
    {
      OpenMS::MzMLFile().transform(path, &result.digest, true, true);
    }
    else if (mode == "load")
    {
      result.retained_experiment = std::make_shared<OpenMS::MSExperiment>();
      auto& experiment = *result.retained_experiment;
      OpenMS::MzMLFile().load(path, experiment);
      for (auto& spectrum : experiment) { result.digest.consumeSpectrum(spectrum); }
      for (auto& chromatogram : experiment.getChromatograms()) { result.digest.consumeChromatogram(chromatogram); }
    }
    else if (mode == "idle")
    {
      // Warmup and worker creation are common; no input/context allocations.
    }
    else if (mode == "contexts" || mode == "contexts-metadata")
    {
      result.retained_context = std::make_shared<OpenMS::OnDiscMSExperiment>();
      if (!result.retained_context->openFile(path, mode != "contexts-metadata"))
      {
        throw std::runtime_error("Input has no usable mzML index");
      }
      result.open_seconds = std::chrono::duration<double>(Clock::now() - begin).count();
      result.index_spectra = result.retained_context->getNrSpectra();
      result.index_chromatograms = result.retained_context->getNrChromatograms();
    }
    else if (mode == "indexed" || mode == "index-open" || mode == "indexed-metadata")
    {
      OpenMS::OnDiscMSExperiment experiment;
      if (!experiment.openFile(path, mode != "indexed-metadata"))
      {
        throw std::runtime_error("Input has no usable mzML index");
      }
      const auto opened = Clock::now();
      result.open_seconds = std::chrono::duration<double>(opened - begin).count();
      result.index_spectra = experiment.getNrSpectra();
      result.index_chromatograms = experiment.getNrChromatograms();
      if (mode == "indexed" || mode == "indexed-metadata")
      {
        if (experiment.getNrSpectra() == 0)
        {
          throw std::runtime_error("Indexed input has no spectra");
        }
        std::mt19937_64 random(20261019);
        for (std::size_t i = 0; i < queries; ++i)
        {
          auto spectrum = experiment.getSpectrum(random() % experiment.getNrSpectra());
          result.digest.consumeSpectrum(spectrum);
        }
      }
      result.read_seconds = std::chrono::duration<double>(Clock::now() - opened).count();
    }
    else
    {
      throw std::runtime_error("Expected stream, load, indexed[-metadata], index-open, contexts[-metadata], or idle");
    }
    if (mode == "stream" || mode == "load" || mode == "idle")
    {
      result.read_seconds = std::chrono::duration<double>(Clock::now() - begin).count();
    }
    return result;
  }

  void generate(const std::string& path, std::size_t count, std::size_t points, bool compressed)
  {
    OpenMS::PlainMSDataWritingConsumer writer(path);
    OpenMS::PeakFileOptions options;
    options.setCompression(compressed);
    options.setWriteIndex(true);
    writer.setOptions(options);
    writer.setExpectedSize(count, 1);
    writer.setExperimentalSettings(OpenMS::ExperimentalSettings());
    std::mt19937 random(20261019);
    for (std::size_t i = 0; i < count; ++i)
    {
      OpenMS::MSSpectrum spectrum;
      spectrum.setNativeID("scan=" + std::to_string(i + 1));
      spectrum.setRT(static_cast<double>(i) * 0.5);
      spectrum.setMSLevel(1);
      spectrum.setType(OpenMS::SpectrumSettings::SpectrumType::CENTROID);
      spectrum.setMetaValue("xml_poc_note", std::string("Unicode \xCE\xB2 & <escaped> \"value\""));
      spectrum.reserve(points);
      for (std::size_t p = 0; p < points; ++p)
      {
        OpenMS::Peak1D peak;
        peak.setMZ(100.0 + static_cast<double>(p) * 0.01 + static_cast<double>(i % 17) * 0.00001);
        peak.setIntensity(static_cast<float>(random() % 1000000) / 8.0F);
        spectrum.push_back(peak);
      }
      writer.consumeSpectrum(spectrum);
    }
    OpenMS::MSChromatogram chromatogram;
    chromatogram.setNativeID("TIC");
    for (std::size_t i = 0; i < count; ++i)
    {
      OpenMS::ChromatogramPeak peak;
      peak.setRT(static_cast<double>(i) * 0.5);
      peak.setIntensity(static_cast<float>(i % 101));
      chromatogram.push_back(peak);
    }
    writer.consumeChromatogram(chromatogram);
  }

  void verify(const std::string& path)
  {
    OpenMS::MSExperiment loaded;
    OpenMS::MzMLFile().load(path, loaded);
    Digest expected;
    for (auto& spectrum : loaded) { expected.consumeSpectrum(spectrum); }
    for (auto& chromatogram : loaded.getChromatograms()) { expected.consumeChromatogram(chromatogram); }
    const auto streamed = run("stream", path, 0).digest;
    OpenMS::OnDiscMSExperiment indexed;
    if (!indexed.openFile(path)) { throw std::runtime_error("Index verification failed"); }
    Digest actual;
    for (std::size_t i = 0; i < indexed.getNrSpectra(); ++i)
    {
      auto spectrum = indexed.getSpectrum(i);
      const auto& original = loaded.getSpectra().at(i);
      if (spectrum.getRT() != original.getRT() || spectrum.getMSLevel() != original.getMSLevel() ||
          spectrum.getFloatDataArrays() != original.getFloatDataArrays() ||
          spectrum.getIntegerDataArrays() != original.getIntegerDataArrays() ||
          spectrum.getStringDataArrays() != original.getStringDataArrays() ||
          spectrum.getMetaValue("xml_poc_note") != original.getMetaValue("xml_poc_note"))
      {
        throw std::runtime_error("Indexed spectrum metadata differ from full load");
      }
      actual.consumeSpectrum(spectrum);
    }
    for (std::size_t i = 0; i < indexed.getNrChromatograms(); ++i)
    {
      auto chromatogram = indexed.getChromatogram(i);
      actual.consumeChromatogram(chromatogram);
    }
    if (!(expected == streamed) || !(expected == actual))
    {
      throw std::runtime_error("Decoded data differ between load, streaming and indexed access");
    }
    std::cout << "{\"verified\":true,\"spectra\":" << expected.spectra
              << ",\"points\":" << expected.points << ",\"hash\":\"" << expected.hash << "\"}\n";
  }
}

{
  std::cerr << "XML_BENCHMARK_MAIN_STARTED\n";

    for (const auto& error : errors) { if (error) { std::rethrow_exception(error); } }
    for (const auto& result : results)

      if (result.index_spectra != results.front().index_spectra ||
          result.index_chromatograms != results.front().index_chromatograms)
      {
        throw std::runtime_error("Worker index count mismatch");
      }
    }
    // Query after joining, while retained contexts/experiments remain alive.
    const auto resident_rss = currentRssBytes();
    std::cout << std::setprecision(12) << "{\"mode\":\"" << mode << "\",\"workers\":" << workers
              << ",\"openms_library\":" << std::quoted(loadedLibrary())
              << ",\"resident_rss_bytes\":" << resident_rss
              << ",\"seconds\":" << seconds << ",\"file_bytes\":" << std::filesystem::file_size(path)
              << ",\"spectra_per_worker\":" << results.front().digest.spectra
              << ",\"chromatograms_per_worker\":" << results.front().digest.chromatograms
              << ",\"points_per_worker\":" << results.front().digest.points
              << ",\"index_spectra\":" << results.front().index_spectra
              << ",\"index_chromatograms\":" << results.front().index_chromatograms
              << ",\"hash_per_worker\":\"" << results.front().digest.hash << "\",\"jobs\":[";

      std::cout << "{\"open_seconds\":" << results[i].open_seconds
                << ",\"read_seconds\":" << results[i].read_seconds << '}';
    }
    std::cout << "]}\n";
  }
  catch (const std::exception& error)
  {
    std::cerr << error.what() << '\n';
    return 1;
  }
  return 0;
}

      * 
      *  Not all conversions are supported yet, due to DataValue using an Int64 as the largest possible integer.
      *  Thus, if the @p value contains a large UInt64, conversion will fail.
      *  Value ranges are currently also not checked, only for XSD types which happen to match the internal representation.
      * 
      *  @param[out] type An XSD type. If the type is not supported, the returned type will be a string
      *  @param[in] value The value in sting format, e.g. "123.34"
      *  @return The Datavalue with the respective type (double, int or string)
      *  @throws Exception::ConversionError if the value does not fit into the internal representation or (for few types) exceeds the XSD specs.
      * 
      */
      static DataValue fromXSDString(const std::string& type, const std::string& value)
      {
        DataValue data_value;
        // float type
        if (type == "xsd:double" || type == "xsd:float" || type == "xsd:decimal")
        {
          data_value = DataValue(StringUtils::toDouble(value));
        }
        // <=32 bit integer types
        else if (type == "xsd:byte" ||          // 8bit signed
                 type == "xsd:int" ||           // 32bit signed
                 type == "xsd:unsignedShort" || // 16bit unsigned
                 type == "xsd:short" ||         // 16bit signed
                 type == "xsd:unsignedByte" || type == "xsd:unsignedInt")
        {
          data_value = DataValue(StringUtils::toInt32(value));
        }
        // 64 bit integer types
        else if (type == "xsd:long" || type == "xsd:unsignedLong" ||       // 64bit signed or unsigned respectively
                 type == "xsd:integer" || type == "xsd:negativeInteger" || // any 'integer' has arbitrary size... but we have to cope with 64bit for now.
                 type == "xsd:nonNegativeInteger" || type == "xsd:nonPositiveInteger" || type == "xsd:positiveInteger")
        {
          data_value = DataValue(StringUtils::toInt64(value)); // internally a signed 64-bit integer. So if someone uses 2^64-1 as value, toInt64() will raise an exception...
        }
        // everything else is treated as a string
        else
        {
          data_value = DataValue(value);
        }
        return data_value;
      }

    void MzMLHandler::populateSpectraWithData_()
    {

      // Whether spectrum should be populated with data
      if (options_.getFillData())
      {
        std::atomic<int> errCount = 0;
        std::string error_message;
        #pragma omp parallel for
        for (SignedSize i = 0; i < (SignedSize)spectrum_data_.size(); i++)
        {
          // parallel exception catching and re-throwing business
          if (!errCount) // no need to parse further if already an error was encountered
          {
          try
          {
            populateSpectraWithData_(spectrum_data_[i].data,
                                      spectrum_data_[i].default_array_length,
                                      options_,
                                      spectrum_data_[i].spectrum);
            if (options_.getSortSpectraByMZ() && !spectrum_data_[i].spectrum.isSorted())
            {
              spectrum_data_[i].spectrum.sortByPosition();
            }
          }

          catch (OpenMS::Exception::BaseException& e)
          {
            #pragma omp critical(MZMLErrorHandling)
            {
              ++errCount;
              error_message = e.what();
            }
          }
          catch (...)
          {
            ++errCount;
            error_message = "Unknown exception during spectrum data population";
          }
        }
        }
        if (errCount != 0)
        {
          std::cerr << "  Parsing error: '" << error_message  << "'" << std::endl;
          std::cerr << "  You could try to disable sorting spectra while loading." << std::endl;
          throw Exception::ParseError(__FILE__, __LINE__, OPENMS_PRETTY_FUNCTION, file_, "Error during parsing of binary data: '" + error_message + "'");
        }
      }

      // Now that binary data arrays have been decoded and materialized into
      // each spectrum's float data arrays, we can reliably check containsIMData().
      // Default any IM spectrum with UNKNOWN peak type to IM_PROFILE.
      for (Size i = 0; i < spectrum_data_.size(); i++)
      {
        auto& spec = spectrum_data_[i].spectrum;
        if (spec.containsIMData() && spec.getIMPeakType() == IMPeakType::UNKNOWN)
        {
          spec.setIMPeakType(IMPeakType::IM_PROFILE);
        }
      }

      // Append all spectra to experiment / consumer
      for (Size i = 0; i < spectrum_data_.size(); i++)
      {
        if (consumer_ != nullptr)
        {
          consumer_->consumeSpectrum(spectrum_data_[i].spectrum);
          if (options_.getAlwaysAppendData())
          {
            exp_->addSpectrum(std::move(spectrum_data_[i].spectrum));
          }
        }
        else
        {
          exp_->addSpectrum(std::move(spectrum_data_[i].spectrum));
        }
      }

      // Delete batch
      spectrum_data_.clear();
    }
